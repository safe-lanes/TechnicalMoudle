Query Logic:

SELECT 
  work_orders.job_code,
  work_orders.job_title,
  components.code as component_code,
  components.name as component_name,
  work_orders.department,
  work_orders.next_due_date,
  (CURRENT_DATE - work_orders.next_due_date) as days_overdue,
  work_orders.maintenance_basis as type,
  work_orders.assigned_to_rank,
  work_orders.class_related
FROM work_orders
JOIN components ON work_orders.component_id = components.id
WHERE work_orders.status = 'Planned'
  AND work_orders.next_due_date < CURRENT_DATE
ORDER BY days_overdue DESC
