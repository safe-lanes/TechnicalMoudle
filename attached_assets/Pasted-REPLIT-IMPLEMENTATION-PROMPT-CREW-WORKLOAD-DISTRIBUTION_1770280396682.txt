// GET /api/reports/crew-workload-distribution
router.get('/crew-workload-distribution', async (req, res) => {
  try {
    const { vessel_id, start_date, end_date, rank, department, view_type } = req.query;
    
    if (view_type === 'summary') {
      // Summary view - Aggregated by rank
      let query = `
        SELECT 
          wo.assigned_to as rank,
          wo.department,
          COUNT(wo.id) as total_jobs_assigned,
          COUNT(CASE WHEN wo.status = 'Completed' THEN 1 END) as jobs_completed,
          COUNT(CASE WHEN wo.status IN ('Planned', 'Active') THEN 1 END) as jobs_pending,
          COUNT(CASE WHEN wo.status = 'Overdue' THEN 1 END) as jobs_overdue,
          COUNT(CASE WHEN wo.critical = 1 THEN 1 END) as critical_jobs,
          COUNT(CASE WHEN wo.job_priority = 'High' THEN 1 END) as high_priority_jobs,
          COALESCE(SUM(wo.manhours), 0) as total_manhours,
          COALESCE(AVG(wo.total_time_taken_hours), 0) as avg_time_per_job,
          ROUND(
            CAST(COUNT(CASE WHEN wo.status = 'Completed' THEN 1 END) AS FLOAT) / 
            NULLIF(COUNT(wo.id), 0) * 100, 
            2
          ) as completion_rate
        FROM work_orders wo
        WHERE wo.vessel_id = ?
      `;
      
      const params = [vessel_id];
      
      if (start_date) {
        query += ` AND wo.created_at >= ?`;
        params.push(start_date);
      }
      
      if (end_date) {
        query += ` AND wo.created_at <= ?`;
        params.push(end_date);
      }
      
      if (rank && rank !== 'All Ranks') {
        query += ` AND wo.assigned_to = ?`;
        params.push(rank);
      }
      
      if (department && department !== 'All') {
        query += ` AND wo.department = ?`;
        params.push(department);
      }
      
      query += ` 
        GROUP BY wo.assigned_to, wo.department
        ORDER BY total_manhours DESC
      `;
      
      const summaryData = db.prepare(query).all(...params);
      
      // Calculate workload percentage
      const totalManhours = summaryData.reduce((sum, row) => sum + row.total_manhours, 0);
      summaryData.forEach(row => {
        row.workload_percentage = totalManhours > 0 
          ? Math.round((row.total_manhours / totalManhours) * 100 * 100) / 100 
          : 0;
      });
      
      res.json({ success: true, data: summaryData, view: 'summary' });
      
    } else {
      // Detailed view - Job-level details
      let query = `
        SELECT 
          wo.id,
          wo.work_order_no,
          wo.job_title,
          c.component_name,
          c.component_code,
          wo.assigned_to,
          wo.performed_by,
          wo.department,
          wo.job_priority,
          wo.status,
          wo.task_type,
          wo.due_date,
          wo.completion_date,
          wo.no_of_persons_in_team as team_size,
          wo.total_time_taken_hours as time_taken_hours,
          wo.manhours,
          wo.critical,
          wo.class_related,
          wo.submitted_date
        FROM work_orders wo
        LEFT JOIN components c ON wo.component_id = c.id
        WHERE wo.vessel_id = ?
      `;
      
      const params = [vessel_id];
      
      if (start_date) {
        query += ` AND wo.created_at >= ?`;
        params.push(start_date);
      }
      
      if (end_date) {
        query += ` AND wo.created_at <= ?`;
        params.push(end_date);
      }
      
      if (rank && rank !== 'All Ranks') {
        query += ` AND wo.assigned_to = ?`;
        params.push(rank);
      }
      
      if (department && department !== 'All') {
        query += ` AND wo.department = ?`;
        params.push(department);
      }
      
      query += ` ORDER BY wo.assigned_to, wo.due_date DESC`;
      
      const detailedData = db.prepare(query).all(...params);
      
      res.json({ success: true, data: detailedData, view: 'detailed' });
    }
    
  } catch (error) {
    console.error('Error fetching crew workload distribution:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

async function generateCrewWorkloadDistributionPDF(data, filters, viewType) {
  const doc = new PDFDocument({ margin: 30, size: 'A4', layout: viewType === 'summary' ? 'portrait' : 'landscape' });
  const chunks = [];
  
  doc.on('data', chunk => chunks.push(chunk));
  
  // Header
  doc.fontSize(18).font('Helvetica-Bold').text('CREW WORKLOAD DISTRIBUTION REPORT', { align: 'center' });
  doc.moveDown(0.5);
  doc.fontSize(10).font('Helvetica');
  doc.text(`Vessel: ${filters.vessel_name || 'All Vessels'}`, { align: 'center' });
  doc.text(`Period: ${filters.start_date || 'All'} to ${filters.end_date || 'All'}`, { align: 'center' });
  doc.text(`Rank: ${filters.rank || 'All Ranks'} | Department: ${filters.department || 'All'}`, { align: 'center' });
  doc.text(`View: ${viewType === 'summary' ? 'Summary by Rank' : 'Detailed Job List'}`, { align: 'center' });
  doc.text(`Generated: ${new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' })}`, { align: 'center' });
  doc.moveDown(1);
  
  if (viewType === 'summary') {
    // Summary View - Table with rank-wise aggregates
    const headers = [
      'S.No',
      'Rank',
      'Dept',
      'Total Jobs',
      'Completed',
      'Pending',
      'Overdue',
      'Manhours',
      'Avg Time',
      'Completion %',
      'Workload %'
    ];
    
    const colWidths = [35, 85, 50, 60, 60, 55, 55, 60, 55, 70, 65];
    let xPos = 30;
    const yStart = doc.y;
    
    // Draw header row with light blue background
    doc.rect(30, yStart, 550, 20).fillAndStroke('#4A90E2', '#000');
    doc.fillColor('#FFF').fontSize(8).font('Helvetica-Bold');
    
    headers.forEach((header, i) => {
      doc.text(header, xPos + 2, yStart + 5, { width: colWidths[i] - 4, align: 'center' });
      xPos += colWidths[i];
    });
    
    doc.moveDown(1.5);
    
    // Table rows
    doc.font('Helvetica').fontSize(8).fillColor('#000');
    
    data.forEach((row, index) => {
      const yPos = doc.y;
      
      // Alternate row colors
      if (index % 2 === 0) {
        doc.rect(30, yPos, 550, 18).fill('#F5F5F5');
      }
      
      xPos = 30;
      const rowData = [
        (index + 1).toString(),
        row.rank || 'Unassigned',
        row.department || 'N/A',
        row.total_jobs_assigned?.toString() || '0',
        row.jobs_completed?.toString() || '0',
        row.jobs_pending?.toString() || '0',
        row.jobs_overdue?.toString() || '0',
        row.total_manhours?.toFixed(1) || '0.0',
        row.avg_time_per_job?.toFixed(1) || '0.0',
        row.completion_rate?.toFixed(1) + '%' || '0.0%',
        row.workload_percentage?.toFixed(1) + '%' || '0.0%'
      ];
      
      rowData.forEach((cell, i) => {
        doc.fillColor('#000').text(cell, xPos + 2, yPos + 4, { 
          width: colWidths[i] - 4, 
          align: i === 0 ? 'center' : (i > 2 ? 'right' : 'left'),
          ellipsis: true
        });
        xPos += colWidths[i];
      });
      
      doc.moveDown(1.2);
      
      // Page break check
      if (doc.y > 700) {
        doc.addPage({ margin: 30, size: 'A4', layout: 'portrait' });
        doc.moveDown(1);
      }
    });
    
    // Summary totals
    const totalJobs = data.reduce((sum, row) => sum + (row.total_jobs_assigned || 0), 0);
    const totalManhours = data.reduce((sum, row) => sum + (row.total_manhours || 0), 0);
    
    doc.moveDown(1);
    doc.fontSize(9).font('Helvetica-Bold').fillColor('#333');
    doc.text(`Total Crew Members: ${data.length}`, 30);
    doc.text(`Total Jobs: ${totalJobs}`);
    doc.text(`Total Manhours: ${totalManhours.toFixed(1)}`);
    
  } else {
    // Detailed View - Landscape with all job details
    const headers = [
      'S.No',
      'Work Order',
      'Component',
      'Job Title',
      'Assigned To',
      'Performed By',
      'Team Size',
      'Time (hrs)',
      'Manhours',
      'Status',
      'Priority',
      'Due Date'
    ];
    
    const colWidths = [30, 75, 80, 100, 70, 70, 50, 50, 55, 60, 55, 65];
    let xPos = 30;
    const yStart = doc.y;
    
    // Header row
    doc.rect(30, yStart, 760, 20).fillAndStroke('#4A90E2', '#000');
    doc.fillColor('#FFF').fontSize(7).font('Helvetica-Bold');
    
    headers.forEach((header, i) => {
      doc.text(header, xPos + 2, yStart + 5, { width: colWidths[i] - 4, align: 'center' });
      xPos += colWidths[i];
    });
    
    doc.moveDown(1.5);
    
    // Data rows
    doc.font('Helvetica').fontSize(7).fillColor('#000');
    
    data.forEach((row, index) => {
      const yPos = doc.y;
      
      if (index % 2 === 0) {
        doc.rect(30, yPos, 760, 16).fill('#F5F5F5');
      }
      
      xPos = 30;
      const rowData = [
        (index + 1).toString(),
        row.work_order_no || 'N/A',
        row.component_name?.substring(0, 20) || 'N/A',
        row.job_title?.substring(0, 25) || 'N/A',
        row.assigned_to || 'N/A',
        row.performed_by || '-',
        row.team_size?.toString() || '-',
        row.time_taken_hours?.toFixed(1) || '-',
        row.manhours?.toFixed(1) || '-',
        row.status || 'N/A',
        row.job_priority || 'N/A',
        row.due_date ? new Date(row.due_date).toLocaleDateString('en-GB') : 'N/A'
      ];
      
      rowData.forEach((cell, i) => {
        doc.fillColor('#000').text(cell, xPos + 2, yPos + 3, { 
          width: colWidths[i] - 4, 
          align: i === 0 ? 'center' : 'left',
          ellipsis: true
        });
        xPos += colWidths[i];
      });
      
      doc.moveDown(1);
      
      if (doc.y > 500) {
        doc.addPage({ margin: 30, size: 'A4', layout: 'landscape' });
        doc.moveDown(1);
      }
    });
  }
  
  // Footer
  doc.fontSize(8).fillColor('#666').text(
    `Total Records: ${data.length}`,
    30,
    doc.page.height - 40,
    { align: 'left' }
  );
  
  doc.end();
  
  return new Promise((resolve) => {
    doc.on('end', () => resolve(Buffer.concat(chunks)));
  });
}
