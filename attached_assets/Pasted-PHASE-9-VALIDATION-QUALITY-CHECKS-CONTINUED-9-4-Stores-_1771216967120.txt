sql
SELECT 
  wo.completed_date as date,
  wo.work_order_number,
  wo.job_title as description,
  u.name as performed_by,
  wo.status,
  wo.completion_notes as remarks
FROM work_orders wo
LEFT JOIN users u ON wo.completed_by = u.id
WHERE wo.job_id = :current_job_id
AND wo.status IN ('COMPLETED', 'CANCELLED')
AND wo.vessel_id = :vessel_id
ORDER BY wo.completed_date DESC
LIMIT 100;
