```sql
WITH critical_components AS (
  SELECT 
    c.id,
    c.component_code,
    c.name,
    c.component_category,
    c.department,
    c.location,
    c.critical,
    c.class_item,
    c.current_cumulative_rh AS running_hours,
    c.is_active
  FROM components c
  WHERE c.vessel_id = $1
    AND c.is_active = true
    AND (c.critical = true OR c.class_item = true)
),
component_wo_stats AS (
  SELECT 
    wo.component_code,
    COUNT(*) FILTER (
      WHERE wo.status IN ('Active', 'Due', 'Due (Grace P)', 'Overdue', 'Pending Approval')
    ) AS total_active_wos,
    COUNT(*) FILTER (
      WHERE wo.status = 'Overdue'
    ) AS overdue_count,
    COUNT(*) FILTER (
      WHERE wo.status IN ('Due', 'Active') 
        AND DATE(wo.next_due_date) <= CURRENT_DATE + INTERVAL '7 days'
    ) AS due_soon_count,
    MIN(wo.next_due_date) AS next_due_date,
    MAX(wo.date_completed) AS last_done_date
  FROM work_orders wo
  WHERE wo.vessel_id = $1
    AND wo.is_active = true
  GROUP BY wo.component_code
)
SELECT 
  cc.component_code,
  cc.name AS component_name,
  cc.component_category,
  cc.department,
  cc.location,
  CASE WHEN cc.critical THEN 'Yes' ELSE 'No' END AS is_critical,
  CASE WHEN cc.class_item THEN 'Yes' ELSE 'No' END AS is_class_item,
  cc.running_hours,
  COALESCE(ws.total_active_wos, 0) AS total_work_orders,
  COALESCE(ws.overdue_count, 0) AS overdue_jobs,
  COALESCE(ws.due_soon_count, 0) AS due_soon_jobs,
  ws.next_due_date,
  ws.last_done_date,
  CASE 
    WHEN ws.next_due_date IS NOT NULL 
    THEN (DATE(ws.next_due_date) - CURRENT_DATE)
    ELSE NULL
  END AS days_until_due
FROM critical_components cc
LEFT JOIN component_wo_stats ws ON cc.component_code = ws.component_code
ORDER BY 
  ws.overdue_count DESC NULLS LAST,
  ws.next_due_date ASC NULLS LAST,
  cc.component_name;
