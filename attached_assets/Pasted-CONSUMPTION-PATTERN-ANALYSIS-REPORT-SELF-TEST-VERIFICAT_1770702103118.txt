Test 1.3: Monthly Aggregation Accuracy

sql
-- Manual verification query:
SELECT 
  DATE_TRUNC('month', timestamp_utc) as month,
  SUM(ABS(qty_change_base)) as total_consumed,
  COUNT(*) as event_count,
  COUNT(DISTINCT part_code) as unique_items
FROM stores_ledger
WHERE vessel_id = ? AND event_type = 'CONSUME'
GROUP BY DATE_TRUNC('month', timestamp_utc)
ORDER BY month;
