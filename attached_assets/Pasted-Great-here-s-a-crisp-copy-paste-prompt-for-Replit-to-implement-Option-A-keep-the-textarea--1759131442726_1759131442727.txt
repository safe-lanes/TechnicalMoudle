### Implementation details

**Data model (unchanged):**

```json
// Immediate
{ "immediateCause": { "unsafeAct": string[], "unsafeCondition": string[] } }
// Root
{ "rootCause": { "individualFactor": string[], "systemFactor": string[] } }
```

**String builders (pseudo/TS):**

```ts
function buildImmediateCauseText(ic: { unsafeAct: string[]; unsafeCondition: string[] }): string {
  const sections: string[] = [];
  if (ic?.unsafeAct?.length) {
    sections.push(
      "UNSAFE ACT",
      ...ic.unsafeAct.map(item => `• ${item}`)
    );
  }
  if (ic?.unsafeCondition?.length) {
    if (sections.length) sections.push(""); // blank line between sections
    sections.push(
      "UNSAFE CONDITION",
      ...ic.unsafeCondition.map(item => `• ${item}`)
    );
  }
  return sections.join("\n");
}

function buildRootCauseText(rc: { individualFactor: string[]; systemFactor: string[] }): string {
  const sections: string[] = [];
  if (rc?.individualFactor?.length) {
    sections.push(
      "INDIVIDUAL FACTOR",
      ...rc.individualFactor.map(item => `• ${item}`)
    );
  }
  if (rc?.systemFactor?.length) {
    if (sections.length) sections.push(""); // blank line between sections
    sections.push(
      "SYSTEM FACTOR",
      ...rc.systemFactor.map(item => `• ${item}`)
    );
  }
  return sections.join("\n");
}
```
