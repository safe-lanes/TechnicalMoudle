1. BACKEND API ENDPOINT:
   Location: server/routes.ts
   Endpoint: /technical/api/reports/unplanned-breakdown-jobs
   
   Query Parameters:
   - vesselId (required)
   - startDate (required)
   - endDate (required)
   
   SQL Query Structure:
   ```sql
   SELECT 
     wo.work_order_no,
     wo.component_code,
     wo.component AS component_name,
     wo.job_title,
     wo.brief_work_description,
     wo.created_at,
     wo.date_completed,
     wo.department,
     wo.job_priority,
     wc.performed_by_rank,
     wc.total_time_taken_hours,
     wc.manhours,
     wc.work_carried_out
   FROM work_orders wo
   LEFT JOIN work_completion wc ON wo.id = wc.work_order_id
   WHERE wo.vessel_id = $1
     AND (
       wo.work_order_type = 'Unplanned'
       OR wo.task_type ILIKE '%Unplanned%'
       OR wo.task_type ILIKE '%Breakdown%'
     )
     AND wo.status = 'Completed'
     AND DATE(wo.created_at) BETWEEN $2 AND $3
   ORDER BY wo.created_at DESC;
