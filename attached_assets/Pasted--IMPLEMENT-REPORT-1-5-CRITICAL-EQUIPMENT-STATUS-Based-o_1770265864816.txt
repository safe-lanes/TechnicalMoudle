app.get('/api/reports/critical-equipment-status', async (req, res) => {
  const { vesselId, startDate, endDate } = req.query;
  
  try {
    const query = `
      WITH critical_components AS (
        SELECT 
          c.id,
          c.component_code,
          c.name AS component_name,
          c.component_category,
          c.department,
          c.location,
          c.critical,
          c.class_item,
          c.is_active,
          c.current_cumulative_rh AS running_hours,
          c.condition_based,
          CASE 
            WHEN c.critical = true AND c.class_item = true THEN 'SOLAS + Class'
            WHEN c.critical = true THEN 'SOLAS Critical'
            WHEN c.class_item = true THEN 'Class Critical'
            ELSE 'N/A'
          END AS critical_type
        FROM components c
        WHERE c.vessel_id = $1
          AND c.is_active = true
          AND (c.critical = true OR c.class_item = true)
      ),
      component_work_orders AS (
        SELECT 
          wo.component_code,
          COUNT(*) FILTER (WHERE wo.status IN ('Active', 'Due', 'Due (Grace P)')) AS total_active_wos,
          COUNT(*) FILTER (WHERE wo.status = 'Overdue') AS overdue_count,
          COUNT(*) FILTER (WHERE wo.due_date <= CURRENT_DATE + INTERVAL '7 days' 
                           AND wo.status IN ('Active', 'Due')) AS due_soon_count,
          MIN(wo.next_due_date) AS next_due_date,
          MAX(wo.date_completed) AS last_done_date,
          STRING_AGG(DISTINCT wo.job_priority, ', ') AS priorities,
          STRING_AGG(DISTINCT wo.job_title, ' | ') AS job_titles
        FROM work_orders wo
        WHERE wo.vessel_id = $1
          AND wo.is_active = true
          AND wo.status != 'Completed'
        GROUP BY wo.component_code
      )
      SELECT 
        cc.*,
        COALESCE(cwo.total_active_wos, 0) AS total_work_orders,
        COALESCE(cwo.overdue_count, 0) AS overdue_jobs,
        COALESCE(cwo.due_soon_count, 0) AS due_soon_jobs,
        cwo.next_due_date,
        cwo.last_done_date,
        cwo.priorities AS job_priorities,
        cwo.job_titles,
        CASE 
          WHEN cwo.overdue_count > 0 THEN 'High Risk'
          WHEN cwo.due_soon_count > 0 THEN 'Medium Risk'
          WHEN cwo.total_active_wos > 0 THEN 'Low Risk'
          ELSE 'No Active Jobs'
        END AS risk_level,
        CASE 
          WHEN cwo.next_due_date IS NOT NULL 
          THEN (DATE(cwo.next_due_date) - CURRENT_DATE)
          ELSE NULL
        END AS days_until_due
      FROM critical_components cc
      LEFT JOIN component_work_orders cwo 
        ON cc.component_code = cwo.component_code
      ORDER BY 
        CASE 
          WHEN cwo.overdue_count > 0 THEN 1
          WHEN cwo.due_soon_count > 0 THEN 2
          ELSE 3
        END,
        cwo.next_due_date ASC NULLS LAST,
        cc.component_name;
    `;
    
    const result = await db.query(query, [vesselId]);
    
    const metadata = {
      total_critical_equipment: result.rows.length,
      solas_critical: result.rows.filter(r => r.critical).length,
      class_critical: result.rows.filter(r => r.class_item).length,
      both_solas_and_class: result.rows.filter(r => r.critical && r.class_item).length,
      high_risk: result.rows.filter(r => r.risk_level === 'High Risk').length,
      medium_risk: result.rows.filter(r => r.risk_level === 'Medium Risk').length,
      total_overdue_jobs: result.rows.reduce((sum, r) => sum + r.overdue_jobs, 0),
      report_date: new Date().toISOString()
    };
    
    res.json({
      success: true,
      data: result.rows,
      metadata
    });
    
  } catch (error) {
    console.error('Error generating critical equipment report:', error);
    res.status(500).json({ 
      success: false, 
      error: error.message 
    });
  }
});
