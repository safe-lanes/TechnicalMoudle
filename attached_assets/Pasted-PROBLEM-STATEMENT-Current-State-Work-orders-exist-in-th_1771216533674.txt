Part C: Spare Parts Consumption (CRITICAL - Currently Missing)
Table: work_order_spares or spare_parts_consumption

Link spare parts to work orders:

sql
-- For each completed work order that used spares
INSERT INTO work_order_spares (work_order_id, spare_part_id, quantity_used, issued_by, issued_date, remarks)
SELECT 
  wo.id as work_order_id,
  sp.id as spare_part_id,
  CASE 
    WHEN wo.job_type = 'Overhaul' THEN FLOOR(RANDOM() * 5) + 1
    WHEN wo.job_type = 'Breakdown' THEN FLOOR(RANDOM() * 3) + 1
    ELSE FLOOR(RANDOM() * 2)
  END as quantity_used,
  store_keeper_user_id,
  wo.completed_date,
  'Used for ' || wo.job_title
FROM work_orders wo
CROSS JOIN LATERAL (
  SELECT sp.id
  FROM spare_parts sp
  WHERE sp.component_id = wo.component_id
  ORDER BY RANDOM()
  LIMIT CASE 
    WHEN wo.job_type = 'Inspection' THEN 1
    WHEN wo.job_type = 'Overhaul' THEN 3
    WHEN wo.job_type = 'Breakdown' THEN 2
    ELSE 1
  END
) sp
WHERE wo.vessel_id = '[Vessel 3 UUID]'
AND wo.status = 'COMPLETED'
AND RANDOM() < 0.6; -- 60% of work orders use spares
