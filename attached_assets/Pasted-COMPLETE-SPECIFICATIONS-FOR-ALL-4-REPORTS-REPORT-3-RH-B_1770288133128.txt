REPORT 3: RH-Based Maintenance Triggers
Report Specifications
Frequency: Daily/Weekly

Priority: HIGH

Est. Time: 1-2 min

Outputs: Preview, PDF (no Excel button visible)

Purpose
Show all components approaching or exceeding their running-hour maintenance thresholds to help plan upcoming RH-based maintenance work.

Calculation Logic
javascript
// For each component with RH-based maintenance
Hours Since Last Maintenance = Current Running Hours - Last Maintenance Hours
Hours to Next Maintenance = Threshold Hours - Hours Since Last Maintenance
Hours Overdue = Hours Since Last Maintenance - Threshold Hours (if negative, it's overdue)

// Alert Bands
Status = 
  Hours to Next < 0 ? "OVERDUE" :
  Hours to Next < (Threshold Hours × 0.10) ? "NEAR DUE" : // Within 10% of interval
  "OK"

// Example:
// Component: Main Engine
// Threshold: 500 hours
// Last Maintenance at: 2500 hours
// Current Reading: 2980 hours
// Hours Since Last: 2980 - 2500 = 480 hours
// Hours to Next: 500 - 480 = 20 hours
// Status: NEAR DUE (20 is < 50, which is 10% of 500)

INNER JOIN running_hours rh ON c.id = rh.component_id
INNER JOIN work_orders wo ON c.id = wo.component_id
WHERE wo.maintenance_basis IN ('Running Hours', 'Mixed')
  AND c.vessel_id = ?
  AND c.is_active = true
  AND wo.is_active = true

REPORT 4: Condition Monitoring Trends
Report Specifications
Frequency: Monthly

Priority: MEDIUM

Est. Time: 3-5 min

Outputs: Preview, PDF, Excel

Purpose
Track equipment condition parameters (vibration, temperature, pressure, lube oil analysis, etc.) over time to identify deteriorating equipment before failure.

Calculation Logic
javascript
// For each component with condition monitoring
const calculateTrends = (readings) => {
    // Time series analysis
    const values = readings.map(r => r.value);
    const dates = readings.map(r => new Date(r.date));
    
    // Statistical metrics
    const min = Math.min(...values);
    const max = Math.max(...values);
    const avg = values.reduce((a, b) => a + b, 0) / values.length;
    const stdDev = Math.sqrt(
        values.reduce((sum, val) => sum + Math.pow(val - avg, 2), 0) / values.length
    );
    
    // Trend direction (linear regression slope)
    const n = values.length;
    const sumX = dates.reduce((sum, d, i) => sum + i, 0);
    const sumY = values.reduce((a, b) => a + b, 0);
    const sumXY = values.reduce((sum, val, i) => sum + (i * val), 0);
    const sumX2 = dates.reduce((sum, d, i) => sum + (i * i), 0);
    
    const slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
    
    const trend = 
        slope > 0.1 ? "Worsening" :
        slope < -0.1 ? "Improving" :
        "Stable";
    
    // Rate of change (per week/month)
    const timeSpan = (dates[n-1] - dates[0]) / (1000 * 60 * 60 * 24); // days
    const changeRate = (values[n-1] - values[0]) / (timeSpan / 30); // per month
    
    // Threshold crossings
    const alertCrossings = readings.filter(r => r.status_band === 'Alert').length;
    const alarmCrossings = readings.filter(r => r.status_band === 'Alarm').length;
    
    return {
        min, max, avg, stdDev,
        trend,
        changeRate,
        latestValue: values[n-1],
        latestStatus: readings[n-1].status_band,
        alertCount: alertCrossings,
        alarmCount: alarmCrossings
    };
};

condition_readings Table Schema:

sql
CREATE TABLE IF NOT EXISTS condition_readings (
  id TEXT PRIMARY KEY,
  component_id TEXT NOT NULL,
  date TIMESTAMP NOT NULL,
  parameter_name TEXT NOT NULL, -- 'vibration', 'temperature', 'pressure', 'oil_analysis', etc.
  value REAL NOT NULL,
  unit TEXT,
  status_band TEXT DEFAULT 'Normal', -- 'Normal', 'Alert', 'Alarm'
  threshold_alert REAL,
  threshold_alarm REAL,
  source TEXT DEFAULT 'manual', -- 'manual', 'sensor', 'lab'
  recorded_by TEXT,
  vessel_id TEXT,
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (component_id) REFERENCES components(id)
);

CREATE INDEX idx_condition_readings_component ON condition_readings(component_id);
CREATE INDEX idx_condition_readings_date ON condition_readings(date);
CREATE INDEX idx_condition_readings_parameter ON condition_readings(parameter_name);
Query Example:

sql
SELECT 
    c.component_name,
    c.component_code,
    c.component_category,
    c.location,
    cr.parameter_name,
    cr.value,
    cr.unit,
    cr.status_band,
    cr.date,
    cr.source
FROM components c
INNER JOIN condition_readings cr ON c.id = cr.component_id
WHERE c.vessel_id = ?
  AND c.condition_based = true
  AND cr.date BETWEEN ? AND ?
ORDER BY c.component_name, cr.parameter_name, cr.date
