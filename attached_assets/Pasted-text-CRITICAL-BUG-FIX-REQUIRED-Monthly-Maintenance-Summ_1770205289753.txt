MAN-HOURS SUMMARY

SELECT 
  SUM(manhours) as total_manhours,
  AVG(manhours) as avg_per_job,
  SUM(CASE WHEN department = 'Engine' THEN manhours ELSE 0 END) as engine_hours,
  SUM(CASE WHEN department = 'Deck' THEN manhours ELSE 0 END) as deck_hours,
  SUM(CASE WHEN department = 'Electrical' THEN manhours ELSE 0 END) as electrical_hours

WHERE wo.vessel_id = @vesselId
