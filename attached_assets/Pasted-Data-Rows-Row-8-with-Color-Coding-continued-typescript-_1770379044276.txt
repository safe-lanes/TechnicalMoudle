function filterCriticalSpares(reportData: CriticalSpareReportRow[]): CriticalSpareReportRow[] {
  return reportData.filter(spare => 
    spare.stockStatus === 'ZERO' ||           // Out of stock
    spare.stockStatus === 'LOW' ||            // Low stock
    spare.criticalityLevel === 'CRITICAL' ||  // Linked to critical equipment
    spare.linkedToCriticalEquipment === true  // Explicitly critical
  );
}
Sort Order (Priority-based):
typescript
function sortReportData(data: CriticalSpareReportRow[]): CriticalSpareReportRow[] {
  return data.sort((a, b) => {
    // Priority 1: Status (ZERO > LOW > OK > NOT_SET)
    const statusPriority = { ZERO: 1, LOW: 2, OK: 3, NOT_SET: 4 };
    const statusDiff = statusPriority[a.stockStatus] - statusPriority[b.stockStatus];
    if (statusDiff !== 0) return statusDiff;
    
    // Priority 2: Criticality (CRITICAL > ESSENTIAL > NORMAL)
    const criticalityPriority = { CRITICAL: 1, ESSENTIAL: 2, NORMAL: 3 };
    const critDiff = criticalityPriority[a.criticalityLevel] - criticalityPriority[b.criticalityLevel];
    if (critDiff !== 0) return critDiff;
    
    // Priority 3: Shortage quantity (DESC - higher shortage first)
    const shortageDiff = b.shortageQty - a.shortageQty;
    if (shortageDiff !== 0) return shortageDiff;
    
    // Priority 4: Part Code (ASC - alphabetical)
    return a.partCode.localeCompare(b.partCode);
  });
}

app.post('/technical/api/reports/critical-spares', async (req, res) => {
  try {
    const { vesselId, filters = {} } = req.body;
    
    if (!vesselId) {
      return res.status(400).json({ error: 'vesselId is required' });
    }
    
    console.log('🔧 Generating Critical Spares Report for vessel:', vesselId);
    
    // Fetch data
    const spares = await storage.getSpares(vesselId);
    const jobs = await storage.getJobs(vesselId);
    const components = await storage.getComponents(vesselId);
    const jobComponentLinks = await storage.getJobComponentLinks(vesselId);
    const vessels = await storage.getVessels();
    
    const vessel = vessels.find(v => v.id === vesselId);
    if (!vessel) {
      return res.status(404).json({ error: 'Vessel not found' });
    }
    
    // Process each spare
    const reportData: CriticalSpareReportRow[] = spares.map((spare, index) => {
      const stockStatus = determineStockStatus(spare);
      const shortage = calculateShortage(spare);
      const criticality = determineCriticality(spare, jobs, jobComponentLinks, components);
      const relatedInfo = getRelatedJobsAndComponents(spare, jobs, jobComponentLinks, components);
      const remarks = generateRemarks(spare, stockStatus, criticality, shortage);
      
      return {
        sNo: index + 1,
        vesselName: vessel.vesselName,
        partCode: spare.partCode,
        partName: spare.partName,
        rob: spare.rob,
        minStock: spare.min,
        stockStatus,
        shortageQty: shortage,
        criticalityLevel: criticality,
        linkedToCriticalEquipment: relatedInfo.linkedToCriticalEquipment,
        criticalComponents: relatedInfo.criticalComponents.join(', '),
        relatedJobs: relatedInfo.relatedJobs.join(', '),
        equipmentCategory: null,  // Can be enhanced later
        department: relatedInfo.department,
        remarks
      };
    });
    
    // Apply filters
    let filteredData = reportData;
    
    if (filters.criticalOnly === true) {
      filteredData = filterCriticalSpares(filteredData);
    }
    
    if (filters.stockStatus && filters.stockStatus.length > 0) {
      filteredData = filteredData.filter(d => 
        filters.stockStatus.includes(d.stockStatus)
      );
    }
    
    if (filters.department) {
      filteredData = filteredData.filter(d => 
        d.department === filters.department
      );
    }
    
    // Sort data
    filteredData = sortReportData(filteredData);
    
    // Re-number after filtering
    filteredData.forEach((row, idx) => {
      row.sNo = idx + 1;
    });
    
    console.log(`📊 Report generated: ${filteredData.length} spares`);
    
    // Generate Excel
    const ExcelJS = require('exceljs');
    const workbook = new ExcelJS.Workbook();
    const worksheet = workbook.addWorksheet('Critical Spares Report');
    
    // [Apply all Excel formatting as described above]
    
    // Send file
    res.setHeader(
      'Content-Type',
      'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
    );
    res.setHeader(
      'Content-Disposition',
      `attachment; filename="${generateFilename('Critical_Spares_Report', vessel.vesselName)}"`
    );
    
    await workbook.xlsx.write(res);
    res.end();
    
  } catch (error) {
    console.error('❌ Error generating critical spares report:', error);
    res.status(500).json({ error: 'Failed to generate report' });
  }
});
