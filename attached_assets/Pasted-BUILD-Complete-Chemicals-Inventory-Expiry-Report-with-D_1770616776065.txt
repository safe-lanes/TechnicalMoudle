// Days Until Expiry
const calculateDaysUntilExpiry = (expiryDate) => {
  if (!expiryDate) return null;
  const today = new Date();
  const expiry = new Date(expiryDate);
  return Math.floor((expiry - today) / (1000 * 60 * 60 * 24));
}

// Expiry Status
const getExpiryStatus = (daysUntilExpiry) => {
  if (daysUntilExpiry === null) return { status: "No Data", color: "gray" };
  if (daysUntilExpiry < 0) return { status: "Expired", color: "red", priority: "Critical" };
  if (daysUntilExpiry <= 30) return { status: "Expiring Soon", color: "red", priority: "Critical" };
  if (daysUntilExpiry <= 60) return { status: "Expiring Soon", color: "orange", priority: "High" };
  if (daysUntilExpiry <= 90) return { status: "Expiring Soon", color: "yellow", priority: "Medium" };
  return { status: "OK", color: "green", priority: "Normal" };
}

// SDS Compliance Percentage
const sdsCompliance = (chemicalsWithSDS / totalChemicals) * 100;

// Auto-calculate expiry from manufacture + shelf life
const calculateExpiryDate = (manufactureDate, shelfLifeMonths) => {
  if (!manufactureDate || !shelfLifeMonths) return null;
  const expiry = new Date(manufactureDate);
  expiry.setMonth(expiry.getMonth() + shelfLifeMonths);
  return expiry;
}

// Summary Counts
const expiredCount = chemicals.filter(c => calculateDaysUntilExpiry(c.expiry_date) < 0).length;
const expiringSoonCount = chemicals.filter(c => {
  const days = calculateDaysUntilExpiry(c.expiry_date);
  return days >= 0 && days <= 90;
}).length;
