Solution A: Backfill Historical Data (Recommended)

-- For each component with current running hours,
-- create a historical baseline entry 30 days ago

INSERT INTO running_hours_log (id, component_id, date, reading, source, vessel_id)
SELECT 
  'BACKFILL-' || c.id || '-' || date('now', '-30 days'),
  c.id,
  date('now', '-30 days'),
  CASE 
    WHEN rh.running_hours > 3000 THEN rh.running_hours - 120  -- Assume 4 hrs/day for high-hour equipment
    WHEN rh.running_hours > 100 THEN rh.running_hours - 20    -- Assume 0.67 hrs/day
    WHEN rh.running_hours > 0 THEN rh.running_hours - 5       -- Assume minimal usage
    ELSE 0
  END as historical_reading,
  'system_estimate',
  c.vessel_id
FROM components c
JOIN running_hours rh ON c.id = rh.component_id
WHERE NOT EXISTS (
  SELECT 1 FROM running_hours_log rhl 
  WHERE rhl.component_id = c.id 
    AND rhl.date <= date('now', '-30 days')
)
AND rh.running_hours > 0;
