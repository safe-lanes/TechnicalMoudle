async function generateCrewWorkloadDistributionExcel(data, filters, viewType) {
  const workbook = new ExcelJS.Workbook();
  const worksheet = workbook.addWorksheet('Crew Workload Distribution');
  
  // Title
  worksheet.mergeCells('A1:L1');
  worksheet.getCell('A1').value = 'CREW WORKLOAD DISTRIBUTION REPORT';
  worksheet.getCell('A1').font = { size: 16, bold: true };
  worksheet.getCell('A1').alignment = { horizontal: 'center' };
  
  // Filter info
  worksheet.mergeCells('A2:L2');
  worksheet.getCell('A2').value = `Vessel: ${filters.vessel_name || 'All'} | Period: ${filters.start_date || 'All'} to ${filters.end_date || 'All'} | Rank: ${filters.rank || 'All Ranks'} | Dept: ${filters.department || 'All'} | View: ${viewType}`;
  worksheet.getCell('A2').alignment = { horizontal: 'center' };
  
  worksheet.addRow([]);
  
  if (viewType === 'summary') {
    // Summary View Headers
    const headerRow = worksheet.addRow([
      'S.No',
      'Rank',
      'Department',
      'Total Jobs Assigned',
      'Jobs Completed',
      'Jobs Pending',
      'Jobs Overdue',
      'Critical Jobs',
      'High Priority Jobs',
      'Total Manhours',
      'Average Time per Job (hrs)',
      'Completion Rate (%)',
      'Workload Distribution (%)'
    ]);
    
    headerRow.font = { bold: true, color: { argb: 'FFFFFFFF' } };
    headerRow.fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FF4A90E2' }
    };
    headerRow.alignment = { horizontal: 'center', vertical: 'middle' };
    
    // Data rows
    data.forEach((row, index) => {
      worksheet.addRow([
        index + 1,
        row.rank || 'Unassigned',
        row.department || 'N/A',
        row.total_jobs_assigned || 0,
        row.jobs_completed || 0,
        row.jobs_pending || 0,
        row.jobs_overdue || 0,
        row.critical_jobs || 0,
        row.high_priority_jobs || 0,
        row.total_manhours ? parseFloat(row.total_manhours.toFixed(2)) : 0,
        row.avg_time_per_job ? parseFloat(row.avg_time_per_job.toFixed(2)) : 0,
        row.completion_rate ? parseFloat(row.completion_rate.toFixed(2)) : 0,
        row.workload_percentage ? parseFloat(row.workload_percentage.toFixed(2)) : 0
      ]);
    });
    
    // Summary totals
    worksheet.addRow([]);
    const totalJobs = data.reduce((sum, row) => sum + (row.total_jobs_assigned || 0), 0);
    const totalCompleted = data.reduce((sum, row) => sum + (row.jobs_completed || 0), 0);
    const totalManhours = data.reduce((sum, row) => sum + (row.total_manhours || 0), 0);
    
    worksheet.addRow(['', 'TOTALS:', '', totalJobs, totalCompleted, '', '', '', '', totalManhours.toFixed(2)]);
    
  } else {
    // Detailed View Headers
    const headerRow = worksheet.addRow([
      'S.No',
      'Work Order No',
      'Component Code',
      'Component Name',
      'Job Title',
      'Assigned To',
      'Performed By',
      'Department',
      'Task Type',
      'Team Size',
      'Time Taken (hrs)',
      'Manhours',
      'Status',
      'Priority',
      'Due Date',
      'Completion Date',
      'Critical',
      'Class Related'
    ]);
    
    headerRow.font = { bold: true, color: { argb: 'FFFFFFFF' } };
    headerRow.fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FF4A90E2' }
    };
    headerRow.alignment = { horizontal: 'center', vertical: 'middle' };
    
    // Data rows
    data.forEach((row, index) => {
      worksheet.addRow([
        index + 1,
        row.work_order_no || 'N/A',
        row.component_code || 'N/A',
        row.component_name || 'N/A',
        row.job_title || 'N/A',
        row.assigned_to || 'N/A',
        row.performed_by || '-',
        row.department || 'N/A',
        row.task_type || 'N/A',
        row.team_size || '-',
        row.time_taken_hours ? parseFloat(row.time_taken_hours.toFixed(2)) : '-',
        row.manhours ? parseFloat(row.manhours.toFixed(2)) : '-',
        row.status || 'N/A',
        row.job_priority || 'N/A',
        row.due_date ? new Date(row.due_date).toLocaleDateString('en-GB') : 'N/A',
        row.completion_date ? new Date(row.completion_date).toLocaleDateString('en-GB') : '-',
        row.critical ? 'Yes' : 'No',
        row.class_related ? 'Yes' : 'No'
      ]);
    });
  }
  
  // Auto-fit columns
  worksheet.columns.forEach((column, index) => {
    if (index === 0) column.width = 8;
    else column.width = 15;
  });
  
  // Apply borders to all data cells
  worksheet.eachRow((row, rowNumber) => {
    if (rowNumber > 3) {
      row.eachCell(cell => {
        cell.border = {
          top: { style: 'thin' },
          left: { style: 'thin' },
          bottom: { style: 'thin' },
          right: { style: 'thin' }
        };
      });
    }
  });
  
  return await workbook.xlsx.writeBuffer();
}

8. AGGREGATION LOGIC
Summary View Calculations:

Total Jobs Assigned = COUNT(*)

Jobs Completed = COUNT(WHERE status = 'Completed')

Jobs Pending = COUNT(WHERE status IN ('Planned', 'Active'))

Jobs Overdue = COUNT(WHERE status = 'Overdue')

Total Manhours = SUM(manhours)

Avg Time per Job = AVG(total_time_taken_hours)

Completion Rate = (Jobs Completed / Total Jobs) × 100

Workload % = (Rank's Manhours / Total Manhours) × 100
