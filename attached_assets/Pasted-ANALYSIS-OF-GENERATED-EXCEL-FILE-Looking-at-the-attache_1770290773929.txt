const getEquipmentUtilizationData = async (vesselId, startDate, endDate, category = null, department = null) => {
  // Step 1: Get all components with RH tracking
  let query = `
    SELECT 
      c.id,
      c.component_code,
      c.component_name,
      c.component_category,
      c.location,
      c.department,
      rh.running_hours AS current_hours,
      rh.last_updated
    FROM components c
    LEFT JOIN running_hours rh ON c.id = rh.component_id
    WHERE c.vessel_id = ?
      AND c.is_active = 1
      AND rh.rh_counter_type != 'Not RH Driven'
  `;
  
  const params = [vesselId];
  
  if (category) {
    query += ` AND c.component_category = ?`;
    params.push(category);
  }
  
  if (department) {
    query += ` AND c.department = ?`;
    params.push(department);
  }
  
  query += ` ORDER BY c.component_code`;
  
  const components = await db.all(query, params);
  
  // Step 2: Calculate period hours for each component
  const daysInPeriod = (new Date(endDate) - new Date(startDate)) / (1000 * 60 * 60 * 24);
  
  const enrichedComponents = await Promise.all(
    components.map(async (comp) => {
      // Get start period reading
      const startReading = await db.get(
        `SELECT running_hours 
         FROM running_hours_log 
         WHERE component_id = ? AND date <= ?
         ORDER BY date DESC LIMIT 1`,
        [comp.id, startDate]
      );
      
      const startHours = startReading?.running_hours || 0;
      const currentHours = comp.current_hours || 0;
      const periodHours = currentHours - startHours;
      const avgDailyHours = daysInPeriod > 0 ? periodHours / daysInPeriod : 0;
      
      // Calculate utilization
      const utilizationBand = 
        avgDailyHours > 20 ? 'High' :
        avgDailyHours >= 10 ? 'Normal' : 'Low';
      
      const utilizationPercent = ((avgDailyHours / 24) * 100).toFixed(1);
      
      return {
        ...comp,
        currentHours,
        periodHours,
        avgDailyHours: avgDailyHours.toFixed(2),
        utilizationBand,
        utilizationPercent,
        lastUpdated: comp.last_updated || null
      };
    })
  );
  
  // Step 3: Calculate summary statistics
  const summary = {
    total: enrichedComponents.length,
    high: enrichedComponents.filter(c => c.utilizationBand === 'High').length,
    normal: enrichedComponents.filter(c => c.utilizationBand === 'Normal').length,
    low: enrichedComponents.filter(c => c.utilizationBand === 'Low').length,
    avgUtilization: (enrichedComponents.reduce((sum, c) => sum + parseFloat(c.utilizationPercent), 0) / enrichedComponents.length).toFixed(1)
  };
  
  return {
    components: enrichedComponents,
    summary,
    vesselName: `Vessel ${vesselId}`,
    startDate,
    endDate,
    daysInPeriod
  };
};
