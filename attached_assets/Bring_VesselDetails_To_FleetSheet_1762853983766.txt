'------------------------------ SUMMARY ------------------------------
'Macro Name : Bring_VesselDetails_To_FleetSheet
'Purpose    : Populate Fleet Component Sheet details from Vessel Component Sheet
'             using "Fleet Equipment Code" as reference.
'
'PROCESS FLOW:
'1. Skip first 3 rows — start from row 4.
'2. Use Column B in "Fleet_Component Sheet" ? Fleet Equipment Code as reference.
'3. Search for same Fleet Equipment Code in "Vessel_Component Sheet" ? Column A.
'4. When a match is found, copy the following fields:
'       Fleet Equipment Name         ? Column C
'       Component Category           ? Column D
'       Maker                        ? Column E
'       Maker Code                   ? Column F
'       Model                        ? Column G
'       Model Code                   ? Column H
'       Location                     ? Column I
'       Rating                       ? Column J
'       Eqpt / System Department     ? Column K
'       Notes                        ? Column L
'       Vessel Code                  ? Column M
'5. If a match is not found, leave cells blank (do not overwrite).
'6. Popup shown once completed: "? Fleet Component Details Updated Successfully!"
'----------------------------------------------------------------------

Sub Bring_VesselDetails_To_FleetSheet()

    Dim wsFleet As Worksheet, wsVessel As Worksheet
    Dim lastRowFleet As Long, lastRowVessel As Long
    Dim i As Long
    Dim fleetCode As String
    Dim foundCell As Range
    
    On Error GoTo ErrHandler
    Application.ScreenUpdating = False
    Application.EnableEvents = False

    '--- Set Sheets ---
    Set wsFleet = ThisWorkbook.Sheets("Fleet_Component Sheet")
    Set wsVessel = ThisWorkbook.Sheets("Vessel_Component Sheet")

    '--- Find Last Rows ---
    lastRowFleet = wsFleet.Cells(wsFleet.Rows.Count, "B").End(xlUp).Row
    lastRowVessel = wsVessel.Cells(wsVessel.Rows.Count, "A").End(xlUp).Row

    '--- Loop through Fleet_Component Sheet ---
    For i = 4 To lastRowFleet
        fleetCode = Trim(wsFleet.Cells(i, "B").Value)
        If fleetCode <> "" Then
            'Search in Vessel_Component Sheet (Column A)
            Set foundCell = wsVessel.Range("A4:A" & lastRowVessel).Find(What:=fleetCode, LookIn:=xlValues, LookAt:=xlWhole)
            
            If Not foundCell Is Nothing Then
                'Copy data from Vessel sheet to Fleet sheet
                wsFleet.Cells(i, "C").Value = wsVessel.Cells(foundCell.Row, "B").Value   'Fleet Equipment Name
                wsFleet.Cells(i, "D").Value = wsVessel.Cells(foundCell.Row, "F").Value   'Component Category
                wsFleet.Cells(i, "E").Value = wsVessel.Cells(foundCell.Row, "G").Value   'Maker
                wsFleet.Cells(i, "F").Value = wsVessel.Cells(foundCell.Row, "H").Value   'Maker Code
                wsFleet.Cells(i, "G").Value = wsVessel.Cells(foundCell.Row, "I").Value   'Model
                wsFleet.Cells(i, "H").Value = wsVessel.Cells(foundCell.Row, "J").Value   'Model Code
                wsFleet.Cells(i, "I").Value = wsVessel.Cells(foundCell.Row, "N").Value   'Location
                wsFleet.Cells(i, "J").Value = wsVessel.Cells(foundCell.Row, "R").Value   'Rating
                wsFleet.Cells(i, "K").Value = wsVessel.Cells(foundCell.Row, "S").Value   'Eqpt / System Department
                wsFleet.Cells(i, "L").Value = wsVessel.Cells(foundCell.Row, "T").Value   'Notes
                wsFleet.Cells(i, "M").Value = wsVessel.Cells(foundCell.Row, "W").Value   'Vessel Code
            End If
        End If
    Next i

    MsgBox "? Fleet Component Details Updated Successfully!", vbInformation

CleanExit:
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Exit Sub

ErrHandler:
    MsgBox "? Error: " & Err.Description, vbCritical
    Resume CleanExit

End Sub
