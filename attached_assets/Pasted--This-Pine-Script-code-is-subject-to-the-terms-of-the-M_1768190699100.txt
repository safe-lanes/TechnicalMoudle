    min = 99999999.
    max = 0.
    idx = 0
    if use_max
        for i = 0 to (bar_index - loc) 
            max := math.max(high[i], max)
            min := max == high[i] ? low[i] : min
            idx := max == high[i] ? i : idx
    else
        for i = 0 to (bar_index - loc) 
            min := math.min(low[i], min)
            max := min == low[i] ? high[i] : max
            idx := min == low[i] ? i : idx

// Function to calculate Dynamic OB Coords
get_ob_coords(bool is_bull, int index) =>
    float res_top = na
    float res_btm = na
    
    // Bullish OB (Demand) - Analyzing Sell Candles (Red) at the Low
    if is_bull 
        bool is_red = close[index] < open[index]
        bool prev_red = close[index + 1] < open[index + 1]
        
        if is_red and prev_red // Cluster
            float max_body = math.max(open[index], close[index])
            float min_body = math.min(open[index], close[index])
            
            int i = 1
            // Check consecutive red candles
            while (index + i < 5000) and (close[index + i] < open[index + i])
                max_body := math.max(max_body, math.max(open[index + i], close[index + i]))
                min_body := math.min(min_body, math.min(open[index + i], close[index + i]))
                i += 1
            res_top := max_body
            res_btm := min_body
        else // Single
            res_top := high[index]
            res_btm := low[index]

    // Bearish OB (Supply) - Analyzing Buy Candles (Green) at the High
    else 
        bool is_green = close[index] > open[index]
        bool prev_green = close[index + 1] > open[index + 1]
        
        if is_green and prev_green // Cluster
            float max_body = math.max(open[index], close[index])
            float min_body = math.min(open[index], close[index])
            
            int i = 1
            // Check consecutive green candles
            while (index + i < 5000) and (close[index + i] > open[index + i])
                max_body := math.max(max_body, math.max(open[index + i], close[index + i]))
                min_body := math.min(min_body, math.min(open[index + i], close[index + i]))
                i += 1
            res_top := max_body
            res_btm := min_body
        else
            res_top := high[index]
            res_btm := low[index]
    
    [res_top, res_btm]

method fnOB(ob[] block, bool bull, float top, float btm, int idx) =>
    // Candle selection fix:
    // - +OB uses last RED candle cluster (same-color), -OB uses last GREEN candle cluster
    // - Doji is ignored
    // - Single-candle OB = High->Low, Multi-candle cluster = BODY bounds only (wicks excluded)
    int   OB_SEARCH   = 25
    int   OB_CLUSTER  = 5
    float DOJI_FRAC   = 0.10

    int anchor   = idx
    int maxSearch = math.min(anchor, OB_SEARCH)
    int obIdx    = na
    bool wantRed = bull

    // Find the nearest expected-color candle between anchor and current
    for s = 0 to maxSearch
        int j = anchor - s
        float rng = high[j] - low[j]
        bool isDoji = not na(rng) and rng > 0 ? math.abs(close[j] - open[j]) <= rng * DOJI_FRAC : true
        bool isRed  = close[j] < open[j]
        bool isGreen = close[j] > open[j]
        bool ok = wantRed ? isRed : isGreen
        if ok and not isDoji
            obIdx := j
            break

    if na(obIdx)
        obIdx := anchor

    // Expand same-color cluster going older from obIdx
    int clen = 1
    for t = 1 to OB_CLUSTER - 1
        int j = obIdx + t
        float rng = high[j] - low[j]
        bool isDoji = not na(rng) and rng > 0 ? math.abs(close[j] - open[j]) <= rng * DOJI_FRAC : true
        bool isRed  = close[j] < open[j]
        bool isGreen = close[j] > open[j]
        bool ok = wantRed ? isRed : isGreen
        if ok and not isDoji
            clen += 1
        else
            break

    float zoneTop = na
    float zoneBtm = na
    float zoneVol = 0.0
    int   locTime = time[obIdx]

    if clen == 1
        zoneTop := high[obIdx]
        zoneBtm := low[obIdx]
        zoneVol := volume[obIdx]
        locTime := time[obIdx]
    else
        // BODY-only range across the cluster
        zoneTop := math.max(open[obIdx], close[obIdx])
        zoneBtm := math.min(open[obIdx], close[obIdx])
        zoneVol := volume[obIdx]
        for t = 1 to clen - 1
            int j = obIdx + t
            zoneTop := math.max(zoneTop, math.max(open[j], close[j]))
            zoneBtm := math.min(zoneBtm, math.min(open[j], close[j]))
            zoneVol += volume[j]
        // Left anchor at the OLDEST candle in the cluster
        locTime := time[obIdx + clen - 1]

    // Fallback to original bounds if anything goes NA
    zoneTop := na(zoneTop) ? top : zoneTop
    zoneBtm := na(zoneBtm) ? btm : zoneBtm

    // Prevent duplicate OBs from being inserted again (causes stacked labels/boxes)
    bool is_duplicate = false
    int  newLoc = locTime
    if block.size() > 0
        int scanTo = math.min(block.size() - 1, 200)
        for k = 0 to scanTo
            ex = block.get(k)
            if ex.loc == newLoc and ex.bull == bull and math.abs(ex.top - zoneTop) <= syminfo.mintick and math.abs(ex.btm - zoneBtm) <= syminfo.mintick
                is_duplicate := true
                break

    if not is_duplicate
        switch bull
            true =>
                blobenter.normal  := false
                blobenter.breaker := false
                array.unshift(block, ob.new(true, zoneTop, zoneBtm, math.avg(zoneTop, zoneBtm), locTime, obupcs, zoneVol, close[obIdx] > open[obIdx] ? 1 : -1, 1, 1, 1, locTime))
            false =>
                brobenter.normal  := false
                brobenter.breaker := false
                array.unshift(block, ob.new(false, zoneTop, zoneBtm, math.avg(zoneTop, zoneBtm), locTime, obdncs, zoneVol, close[obIdx] > open[obIdx] ? 1 : -1, 1, 1, 1, locTime))

        // Hard cap to prevent memory/perf issues in long sessions
        int MAX_OB_STORE = 500
        if block.size() > MAX_OB_STORE
            array.pop(block)

method mitigated(ob[] block) =>
    // Single Source of Truth state-machine:
    //   Fresh OB     : tested=false, isbb=false
    //   Mitigated OB : tested=true , isbb=false   (wick touch only)
    //   Breaker (BR) : isbb=true  (created ONLY on CLOSE invalidation of the OB)
    //   Remove BR    : CLOSE invalidation of the breaker => delete the object
    //
    // Rules:
    //   - Wick touches => mark tested (dashed border), label stays OB
    //   - Close breaks beyond boundary => flip to BR (+/- based on direction)
    //   - Close breaks beyond BR boundary => remove
    //
    // IMPORTANT: use get()+set() so mutations persist.

    if barstate.isconfirmed and block.size() > 0
        for i = block.size() - 1 to 0
            ob z = block.get(i)

            // ---------- ACTIVE ORDER BLOCK ----------
            if not z.isbb
                // 1) Wick touch => mitigate/test (do not flip)
                bool touched = (low <= z.top) and (high >= z.btm)
                if touched
                    z.tested := true

                // 2) Close invalidation => becomes breaker
                if z.bull
                    // +OB invalidates only when CLOSE < bottom
                    if close < z.btm
                        z.isbb  := true
                        z.bbloc := time
                        z.bull  := false         // +OB -> -BR
                        z.tested := false
                else
                    // -OB invalidates only when CLOSE > top
                    if close > z.top
                        z.isbb  := true
                        z.bbloc := time
                        z.bull  := true          // -OB -> +BR
                        z.tested := false

                block.set(i, z)

            // ---------- BREAKER BLOCK ----------
            else
                // Optional: wick touch can mark tested (not required; kept for consistency)
                bool touchedBR = (low <= z.top) and (high >= z.btm)
                if touchedBR
                    z.tested := true

                // Breaker invalidation (CLOSE only) => delete
                bool removeNow = false
                if z.bull
                    // +BR invalidates if CLOSE < bottom
                    removeNow := close < z.btm
                else
                    // -BR invalidates if CLOSE > top
                    removeNow := close > z.top

                if removeNow
                    block.remove(i)
                else
                    block.set(i, z)

    0

overlap(ob[] bull, ob[] bear) =>
    // Overlap removal should apply ONLY to ACTIVE OBs (isbb=false).
    // Never remove / compare Breakers, and never use a Breaker as the "current" reference.

    // -------- helper: find first active OB index --------
    int bullRef = na
    if bull.size() > 0
        for t = 0 to bull.size() - 1
            ob tmp = bull.get(t)
            if not tmp.isbb
                bullRef := t
                break

    int bearRef = na
    if bear.size() > 0
        for t = 0 to bear.size() - 1
            ob tmp = bear.get(t)
            if not tmp.isbb
                bearRef := t
                break

    // -------- bull vs bull --------
    if bull.size() > 1 and not na(bullRef)
        ob current_bull = bull.get(bullRef)
        for i = bull.size() - 1 to 0
            if i == bullRef
                // don't compare ref with itself
                0
            else
                ob stuff_bull = bull.get(i)
                if stuff_bull.isbb or current_bull.isbb
                    0
                else
                    int v = (wichlap == "Recent") ? i : bullRef
                    bool hit =
                         (stuff_bull.btm > current_bull.btm and stuff_bull.btm < current_bull.top) or
                         (stuff_bull.top < current_bull.top and stuff_bull.btm > current_bull.btm) or
                         (stuff_bull.top > current_bull.top and stuff_bull.btm < current_bull.btm) or
                         (stuff_bull.top < current_bull.top and stuff_bull.top > current_bull.btm)
                    if hit and v >= 0 and v < bull.size()
                        ob victim = bull.get(v)
                        if not victim.isbb
                            bull.remove(v)

    // -------- bear vs bear --------
    if bear.size() > 1 and not na(bearRef)
        ob current_bear = bear.get(bearRef)
        for i = bear.size() - 1 to 0
            if i == bearRef
                0
            else
                ob stuff_bear = bear.get(i)
                if stuff_bear.isbb or current_bear.isbb
                    0
                else
                    int v = (wichlap == "Recent") ? i : bearRef
                    bool hit =
                         (stuff_bear.btm > current_bear.btm and stuff_bear.btm < current_bear.top) or
                         (stuff_bear.top < current_bear.top and stuff_bear.btm > current_bear.btm) or
                         (stuff_bear.top > current_bear.top and stuff_bear.btm < current_bear.btm) or
                         (stuff_bear.top < current_bear.top and stuff_bear.top > current_bear.btm)
                    if hit and v >= 0 and v < bear.size()
                        ob victim = bear.get(v)
                        if not victim.isbb
                            bear.remove(v)

    // -------- cross overlap: bull vs bear (active only) --------
    if bull.size() > 0 and bear.size() > 0 and not na(bullRef) and not na(bearRef)
        ob cb = bull.get(bullRef)
        ob cr = bear.get(bearRef)

        // remove active bull that overlaps current active bear
        for i = bull.size() - 1 to 0
            ob s = bull.get(i)
            if not s.isbb and not cr.isbb
                bool hit =
                     (s.btm > cr.btm and s.btm < cr.top) or
                     (s.top < cr.top and s.btm > cr.btm) or
                     (s.top > cr.top and s.btm < cr.btm) or
                     (s.top < cr.top and s.top > cr.btm)
                if hit
                    bull.remove(i)

        // remove active bear that overlaps current active bull
        for i = bear.size() - 1 to 0
            ob s = bear.get(i)
            if not s.isbb and not cb.isbb
                bool hit =
                     (s.btm > cb.btm and s.btm < cb.top) or
                     (s.top < cb.top and s.btm > cb.btm) or
                     (s.top > cb.top and s.btm < cb.btm) or
                     (s.top < cb.top and s.top > cb.btm)
                if hit
                    bear.remove(i)

overlapFVG(FVG[] bull, FVG[] bear) =>
    // Hide overlapping FVGs between bullish and bearish arrays (keeps the most recent).
    // NOTE: We must guard against empty arrays because `for i = size()-1 to 0` will start at -1 when size() == 0.
    if bull.size() == 0 or bear.size() == 0
        0
    else
        for i = bull.size() - 1 to 0
            if i < bull.size()
                FVG a = bull.get(i)
                bool removedA = false

                if bear.size() > 0
                    for j = bear.size() - 1 to 0
                        if j < bear.size()
                            FVG b = bear.get(j)
                            bool hit = a.btm < b.top and a.top > b.btm
                            if hit
                                // remove older of the two
                                if a.loc < b.loc
                                    bull.remove(i)
                                    removedA := true
                                    break
                                else
                                    bear.remove(j)

                if removedA
                    0
        0

    0

f_trim_div(line[] lns, label[] la, label[] lb, int keep) =>
    if keep > 0
        while array.size(lns) > keep
            line _ln = array.shift(lns)
            if not na(_ln)
                line.delete(_ln)
            if array.size(la) > 0
                label _la = array.shift(la)
                if not na(_la)
                    label.delete(_la)
            if array.size(lb) > 0
                label _lb = array.shift(lb)
                if not na(_lb)
                    label.delete(_lb)

// • Draws 0/25/50/75/100 horizontal levels as LINES starting from the *left-most* HTF extreme time.
// • Optional right-side tags (price + level %) — controlled by "Show Right Tags".

// --- HTF extreme helpers (SAFE: no negative indexing) ---
// ta.highestbars/ta.lowestbars return NEGATIVE offsets (e.g., -3). Convert to positive index using unary minus.

f_levels_wick(_len) =>
    int n = _len
    n := n > bar_index + 1 ? bar_index + 1 : n
    n := n < 1 ? 1 : n
    float H = ta.highest(high, n)
    float L = ta.lowest(low, n)
    int hb = nz(-ta.highestbars(high, n), 0)
    int lb = nz(-ta.lowestbars(low, n), 0)
    hb := hb < 0 ? 0 : hb > n - 1 ? n - 1 : hb
    lb := lb < 0 ? 0 : lb > n - 1 ? n - 1 : lb
    int tH = time[hb]
    int tL = time[lb]
    [H, L, tH, tL]

f_levels_close(_len) =>
    int n = _len
    n := n > bar_index + 1 ? bar_index + 1 : n
    n := n < 1 ? 1 : n
    float H = ta.highest(close, n)
    float L = ta.lowest(close, n)
    int hb = nz(-ta.highestbars(close, n), 0)
    int lb = nz(-ta.lowestbars(close, n), 0)
    hb := hb < 0 ? 0 : hb > n - 1 ? n - 1 : hb
    lb := lb < 0 ? 0 : lb > n - 1 ? n - 1 : lb
    int tH = time[hb]
    int tL = time[lb]
    [H, L, tH, tL]
