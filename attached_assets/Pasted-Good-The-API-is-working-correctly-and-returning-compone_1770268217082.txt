async function generateCriticalEquipmentExcel(data, metadata, filters) {
  const ExcelJS = require('exceljs');
  const workbook = new ExcelJS.Workbook();
  const worksheet = workbook.addWorksheet('Critical Equipment Status');
  
  // HEADER (Row 1)
  worksheet.mergeCells('A1:L1');
  worksheet.getCell('A1').value = 'CRITICAL EQUIPMENT STATUS REPORT';
  worksheet.getCell('A1').font = { size: 16, bold: true, color: { argb: 'FFFFFF' } };
  worksheet.getCell('A1').fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: '4472C4' }
  };
  worksheet.getCell('A1').alignment = { horizontal: 'center', vertical: 'middle' };
  worksheet.getRow(1).height = 30;
  
  // SUMMARY (Rows 3-6)
  worksheet.getCell('A3').value = 'Summary:';
  worksheet.getCell('A3').font = { bold: true };
  
  worksheet.getCell('A4').value = `Total Critical Equipment: ${metadata.total_critical_equipment}`;
  worksheet.getCell('A5').value = `Critical Only: ${metadata.critical_only} | Class Item Only: ${metadata.class_item_only} | Both: ${metadata.both_critical_and_class}`;
  worksheet.getCell('A6').value = `Equipment with Overdue: ${metadata.equipment_with_overdue} | Due Soon: ${metadata.equipment_due_soon}`;
  
  // COLUMN HEADERS (Row 8)
  const headers = [
    'S.No', 'Component Code', 'Component Name', 'Critical', 'Class Item',
    'Department', 'Location', 'Total WOs', 'Overdue', 'Due Soon', 'Next Due', 'Days'
  ];
  
  worksheet.getRow(8).values = headers;
  worksheet.getRow(8).font = { bold: true, color: { argb: 'FFFFFF' } };
  worksheet.getRow(8).fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: '4472C4' }
  };
  worksheet.getRow(8).alignment = { horizontal: 'center', vertical: 'middle' };
  worksheet.getRow(8).height = 25;
  
  // DATA ROWS (Starting row 9)
  data.forEach((row, index) => {
    const rowData = [
      index + 1,
      row.component_code || 'N/A',
      row.component_name || 'N/A',
      row.is_critical || 'No',
      row.is_class_item || 'No',
      row.department || 'N/A',
      row.location || 'N/A',
      row.total_work_orders || 0,
      row.overdue_jobs || 0,
      row.due_soon_jobs || 0,
      row.next_due_date ? new Date(row.next_due_date).toLocaleDateString() : 'N/A',
      row.days_until_due !== null ? row.days_until_due : 'N/A'
    ];
    
    const rowIndex = 9 + index;
    worksheet.addRow(rowData);
    
    // CONDITIONAL FORMATTING
    const excelRow = worksheet.getRow(rowIndex);
    
    // Red background for equipment with overdue jobs
    if (row.overdue_jobs > 0) {
      excelRow.fill = {
        type: 'pattern',
        pattern: 'solid',
        fgColor: { argb: 'FFE6E6' }
      };
    }
    // Yellow background for equipment due soon (no overdue)
    else if (row.due_soon_jobs > 0) {
      excelRow.fill = {
        type: 'pattern',
        pattern: 'solid',
        fgColor: { argb: 'FFF9E6' }
      };
    }
    // Alternating gray for others
    else if (index % 2 === 0) {
      excelRow.fill = {
        type: 'pattern',
        pattern: 'solid',
        fgColor: { argb: 'F2F2F2' }
      };
    }
    
    // Bold red text for overdue column if > 0
    const overdueCell = worksheet.getCell(`I${rowIndex}`);
    if (row.overdue_jobs > 0) {
      overdueCell.font = { color: { argb: 'FF0000' }, bold: true };
    }
    
    // Bold red/orange text for days column
    const daysCell = worksheet.getCell(`L${rowIndex}`);
    if (row.days_until_due !== null) {
      if (row.days_until_due < 0) {
        daysCell.font = { color: { argb: 'FF0000' }, bold: true };
      } else if (row.days_until_due <= 7) {
        daysCell.font = { color: { argb: 'FFA500' } };
      }
    }
    
    // Center align all cells
    excelRow.alignment = { horizontal: 'center', vertical: 'middle' };
    
    // Borders
    excelRow.eachCell((cell) => {
      cell.border = {
        top: { style: 'thin' },
        left: { style: 'thin' },
        bottom: { style: 'thin' },
        right: { style: 'thin' }
      };
    });
  });
  
  // COLUMN WIDTHS
  worksheet.getColumn(1).width = 8;   // S.No
  worksheet.getColumn(2).width = 15;  // Code
  worksheet.getColumn(3).width = 30;  // Name
  worksheet.getColumn(4).width = 10;  // Critical
  worksheet.getColumn(5).width = 12;  // Class Item
  worksheet.getColumn(6).width = 15;  // Dept
  worksheet.getColumn(7).width = 15;  // Location
  worksheet.getColumn(8).width = 10;  // Total WOs
  worksheet.getColumn(9).width = 10;  // Overdue
  worksheet.getColumn(10).width = 10; // Due Soon
  worksheet.getColumn(11).width = 15; // Next Due
  worksheet.getColumn(12).width = 8;  // Days
  
  // DOWNLOAD
  const buffer = await workbook.xlsx.writeBuffer();
  const blob = new Blob([buffer], { 
    type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' 
  });
  const url = window.URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = `Critical_Equipment_Status_${new Date().toISOString().split('T')[0]}.xlsx`;
  link.click();
  window.URL.revokeObjectURL(url);
}
