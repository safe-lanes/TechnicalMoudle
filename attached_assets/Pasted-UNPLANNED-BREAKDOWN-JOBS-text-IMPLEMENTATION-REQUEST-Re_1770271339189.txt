app.get('/api/reports/unplanned-breakdown-jobs', async (req, res) => {
  const { vesselId, startDate, endDate } = req.query;
  
  try {
    const query = `
      SELECT 
        wo.work_order_no,
        wo.component_code,
        wo.component AS component_name,
        wo.job_title,
        wo.brief_work_description,
        wo.created_at,
        wo.date_completed,
        wo.submitted_date,
        wc.start_date,
        wc.completion_date,
        wc.performed_by_rank,
        wc.total_time_taken_hours,
        wc.manhours,
        wc.work_carried_out,
        wo.department,
        wo.job_priority
      FROM work_orders wo
      LEFT JOIN work_completion wc ON wo.id = wc.work_order_id
      WHERE wo.vessel_id = $1
        AND (
          wo.work_order_type = 'Unplanned' 
          OR wo.task_type ILIKE '%Unplanned%'
          OR wo.task_type ILIKE '%Breakdown%'
        )
        AND wo.date_completed IS NOT NULL
        AND DATE(wo.created_at) BETWEEN $2 AND $3
      ORDER BY wo.created_at DESC;
    `;
    
    const result = await db.query(query, [vesselId, startDate, endDate]);
    
    const metadata = {
      total_unplanned_jobs: result.rows.length,
      total_manhours: result.rows.reduce((sum, r) => sum + (r.manhours || 0), 0),
      avg_time_taken: result.rows.length > 0 
        ? (result.rows.reduce((sum, r) => sum + (r.total_time_taken_hours || 0), 0) / result.rows.length).toFixed(2)
        : 0,
      by_department: groupByDepartment(result.rows),
      by_priority: groupByPriority(result.rows),
      report_date: new Date().toISOString(),
      date_range: { start: startDate, end: endDate }
    };
    
    res.json({
      success: true,
      data: result.rows,
      metadata
    });
    
  } catch (error) {
    console.error('Error generating unplanned/breakdown jobs report:', error);
    res.status(500).json({ 
      success: false, 
      error: error.message 
    });
  }
});

// Helper functions
function groupByDepartment(rows) {
  return rows.reduce((acc, row) => {
    const dept = row.department || 'Not Specified';
    acc[dept] = (acc[dept] || 0) + 1;
    return acc;
  }, {});
}

function groupByPriority(rows) {
  return rows.reduce((acc, row) => {
    const priority = row.job_priority || 'Not Specified';
    acc[priority] = (acc[priority] || 0) + 1;
    return acc;
  }, {});
}

export async function generateUnplannedBreakdownPDF(
  data: any[], 
  metadata: any,
  filters: any
) {
  const doc = new jsPDF({ 
    orientation: 'landscape', 
    unit: 'mm', 
    format: 'a4' 
  });
  
  const pageWidth = doc.internal.pageSize.getWidth();
  
  // HEADER (same blue as Report 1.5)
  doc.setFillColor(68, 114, 196);
  doc.rect(0, 0, pageWidth, 25, 'F');
  
  doc.setTextColor(255, 255, 255);
  doc.setFontSize(18);
  doc.setFont('helvetica', 'bold');
  doc.text('UNPLANNED/BREAKDOWN JOBS REPORT', pageWidth / 2, 12, { align: 'center' });
  
  doc.setFontSize(10);
  doc.setFont('helvetica', 'normal');
  doc.text(`Generated: ${new Date().toLocaleString()}`, pageWidth / 2, 18, { align: 'center' });

  let yPos = 35;
  doc.setTextColor(0, 0, 0);
  doc.setFontSize(12);
  doc.setFont('helvetica', 'bold');
  doc.text('Summary:', 15, yPos);
  
  yPos += 8;
  doc.setFontSize(10);
  doc.setFont('helvetica', 'normal');
  doc.text(`Date Range: ${filters.startDate} to ${filters.endDate}`, 15, yPos);
  yPos += 6;
  doc.text(`Total Unplanned Jobs: ${metadata.total_unplanned_jobs}`, 15, yPos);
  yPos += 6;
  doc.text(`Total Man Hours: ${metadata.total_manhours}`, 15, yPos);
  yPos += 6;
  doc.text(`Average Time Taken: ${metadata.avg_time_taken} hours`, 15, yPos);
  
  // TABLE
  yPos += 8;
  
  const tableData = data.map((row, index) => [

    row.total_time_taken_hours?.toString() || '0',
    row.manhours?.toString() || '0',
    row.department || 'N/A'
  ]);

    startY: yPos,
    head: [[
      'S.No',
      'WO Number',
      'Comp. Code',
      'Component Name',
      'Job Title',
      'Description',
      'Created Date',
      'Completed Date',
      'Performed By',
      'Hours',
      'Manhours',
      'Dept'
    ]],
    body: tableData,
    theme: 'grid',
    headStyles: {
      fillColor: ,
      textColor: 255,
      fontStyle: 'bold',
      fontSize: 9,
      halign: 'center'
    },
    bodyStyles: {
      fontSize: 8,
      cellPadding: 2
    },
    columnStyles: {
      0: { halign: 'center', cellWidth: 10 },
      1: { halign: 'center', cellWidth: 22 },
      2: { halign: 'center', cellWidth: 15 },
      3: { halign: 'left', cellWidth: 30 },
      4: { halign: 'left', cellWidth: 25 },

      fillColor:

  // FOOTER (page numbers - same as Report 1.5)
