B. Tool Execution Logic
In chatbotService.ts, implement:

typescript
async function executeTool(
  toolName: string, 
  args: any, 
  storage: any
): Promise<any> {
  
  switch (toolName) {
    case 'getworkorders':
      const workOrders = await storage.getWorkOrders(args.vesselId);
      let filtered = workOrders;
      
      // Apply status filter
      if (args.status) {
        filtered = filtered.filter(wo => wo.status === args.status);
      }
      
      // Apply component filter
      if (args.componentCode) {
        filtered = filtered.filter(wo => 
          wo.componentCode?.includes(args.componentCode)
        );
      }
      
      // Apply date range filter
      if (args.dateRange) {
        const now = new Date();
        let cutoffDate = new Date();
        
        if (args.dateRange === 'week') {
          cutoffDate.setDate(now.getDate() + 7);
        } else if (args.dateRange === 'month') {
          cutoffDate.setDate(now.getDate() + 30);
        } else if (args.dateRange === 'quarter') {
          cutoffDate.setDate(now.getDate() + 90);
        }
        
        filtered = filtered.filter(wo => {
          const dueDate = new Date(wo.dueDate);
          return dueDate <= cutoffDate;
        });
      }
      
      // Apply limit
      const limit = args.limit || 50;
      return filtered.slice(0, limit);

    case 'getworkordercounts':
      const allWOs = await storage.getWorkOrders(args.vesselId);
      return {
        total: allWOs.length,
        overdue: allWOs.filter(wo => wo.status === 'OVERDUE').length,
        due: allWOs.filter(wo => wo.status === 'DUE_GRACE_PERIOD').length,
        completed: allWOs.filter(wo => wo.status === 'COMPLETED').length,
        open: allWOs.filter(wo => wo.status === 'OPEN').length
      };

    case 'getlowstockspares':
      const spares = await storage.getSpares(args.vesselId);
      const lowStock = spares.filter(sp => 
        sp.rob !== null && sp.min !== null && sp.rob < sp.min
      );
      
      if (args.criticalOnly) {
        return lowStock.filter(sp => sp.critical === true);
      }
      return lowStock;

    case 'generatedeeplink':
      const baseUrl = '/pms';
      const queryParams = new URLSearchParams(args.filters || {}).toString();
      return {
        url: `${baseUrl}/${args.page}${queryParams ? '?' + queryParams : ''}`,
        displayText: `View ${args.page.replace('-', ' ')}`
      };

    // Add remaining cases for other 7 tools
    
    default:
      throw new Error(`Unknown tool: ${toolName}`);
  }
}

D. Error Handling Patterns
typescript
// In chatbotService.ts
try {
  const response = await openai.chat.completions.create({...});
  // Process response
} catch (error) {
  console.error('OpenAI API error:', error);
  
  if (error.code === 'insufficient_quota') {
    return {
      response: "I'm unable to process your request due to API quota limits. Please contact your system administrator.",
      toolsUsed: []
    };
  }
  
  return {
    response: "I'm having trouble connecting to my AI service right now. Please try again in a moment, or contact support if this persists.",
    toolsUsed: []
  };
}

// In chatbot.ts route
try {
  const result = await processChatMessage(...);
  res.json(result);
} catch (error) {
  console.error('Chat endpoint error:', error);
  res.status(500).json({ 
    error: 'Failed to process chat message',
    message: error.message 
  });
}
