Enhancement 1: Weighted Average Monthly Consumption
Current Implementation:

text
Avg Monthly = (Total Consumed / Days of Data) × 30
Example: 1 pcs consumed in 1 day = 30 pcs/month
Issue with Limited Data:
When data period is < 7 days, this linear projection can be misleading:

1 consumption event on Day 1 doesn't mean 30 events will occur in 30 days

Consumption is typically sporadic, not daily

Suggested Enhancement:
For data periods < 30 days, apply a confidence multiplier to avoid over-projection:

typescript
function calculateAvgMonthlyConsumption(totalConsumed, daysOfData) {
  const rawMonthlyRate = (totalConsumed / daysOfData) * 30;
  
  // Apply confidence multiplier for limited data
  if (daysOfData < 7) {
    // Very limited data - reduce projection by 50%
    return rawMonthlyRate * 0.5;
  } else if (daysOfData < 30) {
    // Partial month - reduce projection by 25%
    return rawMonthlyRate * 0.75;
  } else {
    // Full month or more - use actual rate
    return rawMonthlyRate;
  }
}

function calculateReorderQuantity(item) {
  const { currentROB, minStock, maxStock, avgMonthlyConsumption, leadTimeDays = 30 } = item;
  
  // Calculate safety stock (1 month buffer for critical items)
  const safetyStock = avgMonthlyConsumption;
  
  // Calculate reorder point
  const reorderPoint = (avgMonthlyConsumption / 30 * leadTimeDays) + safetyStock;
  
  // If below reorder point, calculate order quantity
  if (currentROB <= reorderPoint) {
    // Order up to max stock level (or 6 months supply if no max)
    const targetLevel = maxStock || (avgMonthlyConsumption * 6);
    const suggestedQty = Math.ceil(targetLevel - currentROB);
    
    return {
      suggestedQty,
      reorderPoint,
      reasoning: `Bring stock from ${currentROB} to ${targetLevel} (6 months supply)`
    };
  }
  
  return { suggestedQty: 0, reason: 'Stock adequate' };
}

function calculateStockoutEstimate(currentROB, avgDailyConsumption, daysOfData) {
  if (avgDailyConsumption === 0) {
    return { estimate: '∞', range: null };
  }
  
  const baseEstimate = currentROB / avgDailyConsumption;
  
  // Calculate confidence interval based on data quality
  if (daysOfData < 7) {
    // Very limited data - wide range
    const lowerBound = Math.floor(baseEstimate * 0.5);
    const upperBound = Math.ceil(baseEstimate * 2.0);
    return {
      estimate: Math.round(baseEstimate),
      range: `${lowerBound}-${upperBound} days`,
      confidence: 'low'
    };
  } else if (daysOfData < 30) {
    // Moderate data - narrower range
    const lowerBound = Math.floor(baseEstimate * 0.75);
    const upperBound = Math.ceil(baseEstimate * 1.5);
    return {
      estimate: Math.round(baseEstimate),
      range: `${lowerBound}-${upperBound} days`,
      confidence: 'medium'
    };
  } else {
    // Good data - point estimate
    return {
      estimate: Math.round(baseEstimate),
      range: null,
      confidence: 'high'
    };
  }
}
