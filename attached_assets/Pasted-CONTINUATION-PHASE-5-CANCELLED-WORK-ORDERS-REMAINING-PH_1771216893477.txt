CONTINUATION: PHASE 5 - CANCELLED WORK ORDERS & REMAINING PHASES
sql
UPDATE work_orders
SET 
  status = 'CANCELLED',
  cancelled_date = created_date + INTERVAL '5 days',
  cancelled_by = chief_engineer_id,
  cancellation_reason = CASE 
    WHEN RANDOM() < 0.4 THEN 'Equipment replaced/decommissioned'
    WHEN RANDOM() < 0.65 THEN 'Manufacturer recommendation superseded'
    WHEN RANDOM() < 0.85 THEN 'Duplicate work order created in error'
    ELSE 'Job not required based on inspection findings'
  END,
  cancellation_notes = CASE cancellation_reason
    WHEN 'Equipment replaced/decommissioned' THEN 'Component was replaced with new unit. This maintenance job is no longer applicable. Component decommissioning documented separately.'
    WHEN 'Manufacturer recommendation superseded' THEN 'New service bulletin received from manufacturer with revised maintenance schedule. This job cancelled and replaced with updated job template per SB-' || FLOOR(RANDOM() * 9000 + 1000)
    WHEN 'Duplicate work order created in error' THEN 'Duplicate entry. Work order #' || (SELECT work_order_number FROM work_orders wo2 WHERE wo2.component_id = work_orders.component_id AND wo2.id < work_orders.id LIMIT 1) || ' already covers this maintenance.'
    ELSE 'Detailed inspection revealed component condition is satisfactory. Maintenance frequency adjusted. Job rescheduled per revised interval.'
  END
WHERE vessel_id = '[Vessel 3 UUID]'
AND id IN (
  SELECT id FROM work_orders 
  WHERE vessel_id = '[Vessel 3 UUID]' 
  AND status = 'OPEN' 
  ORDER BY RANDOM() 
  LIMIT 25
);

7.1: Link Survey Findings to Rectification Work Orders
sql
UPDATE survey_findings sf
SET rectification_work_order_id = (
  SELECT wo.id
  FROM work_orders wo
  WHERE wo.vessel_id = sf.vessel_id
  AND wo.created_date BETWEEN sf.finding_date AND sf.finding_date + INTERVAL '30 days'
  AND wo.component_id IN (
    SELECT c.id FROM components c 
    WHERE sf.finding_description ILIKE '%' || c.name || '%'
  )
  ORDER BY wo.created_date ASC
  LIMIT 1
)
WHERE sf.vessel_id = '[Vessel 3 UUID]'
AND sf.severity IN ('MAJOR', 'OBSERVATION')
AND sf.rectification_work_order_id IS NULL;
7.2: Update Survey Finding Status Based on Work Order Completion
sql
UPDATE survey_findings sf
SET 
  status = CASE 
    WHEN wo.status = 'COMPLETED' AND wo.completed_date IS NOT NULL THEN 'CLOSED'
    WHEN wo.status IN ('IN_PROGRESS', 'OPEN') THEN 'UNDER_RECTIFICATION'
    ELSE 'PENDING'
  END,
  rectified_date = wo.completed_date,
  verified_date = wo.approved_date,
  verified_by = wo.approved_by
FROM work_orders wo
WHERE sf.rectification_work_order_id = wo.id
AND sf.vessel_id = '[Vessel 3 UUID]';
