router.get('/critical-equipment-status', async (req, res) => {
  try {
    const { vessel_id, date_range } = req.query;
    
    const query = `
      SELECT 
        c.code AS component_code,
        c.name AS component_name,
        c.category AS component_category,
        c.criticality,
        c.class_item,
        c.location,
        c.department,
        c.running_hours,
        c.condition_based,
        c.is_active,
        wo.job_code AS work_order_no,
        wo.job_title,
        wo.next_due_date,
        CAST((wo.next_due_date - CURRENT_DATE) AS INTEGER) AS days_until_due,
        wo.job_priority,
        wo.status,
        wo.class_related,
        wo.frequency_value,
        wo.frequency_unit,
        wo.assigned_to_rank
      FROM components c
      LEFT JOIN work_orders wo ON c.code = wo.component_code
      WHERE (c.criticality = 'Yes' OR c.class_item = 'Yes')
        AND c.is_active = 'Yes'
        AND wo.status = 'Planned'
      ORDER BY wo.next_due_date ASC NULLS LAST, c.code ASC
    `;
    
    const results = await db.query(query, [vessel_id]);
    
    res.json({
      success: true,
      data: results.rows,
      metadata: {
        total_critical_equipment: results.rows.length,
        solas_critical: results.rows.filter(r => r.criticality === 'Yes').length,
        class_critical: results.rows.filter(r => r.class_item === 'Yes').length
      }
    });
    
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
