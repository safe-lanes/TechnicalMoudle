// Add this function to extract suggestions from tool results
const getSuggestedFollowUps = (lastMessage: Message) => {
  if (!lastMessage || lastMessage.role !== 'assistant') return [];
  
  const content = lastMessage.content.toLowerCase();
  
  if (content.includes('overdue')) {
    return [
      'Show me details of the most overdue item',
      'Which system has the most overdue work?',
      'What spares do I need for these jobs?'
    ];
  }
  
  if (content.includes('low stock') || content.includes('spares')) {
    return [
      'Which are critical?',
      'Draft a requisition list',
      'Show me Main Engine spares only'
    ];
  }
  
  if (content.includes('component')) {
    return [
      'Show work orders for this equipment',
      'What are the running hours?',
      'Any related spares low in stock?'
    ];
  }
  
  return [];
};
