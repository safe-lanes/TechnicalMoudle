🧮 CALCULATION LOGIC & BUSINESS RULES
1. Stock Status Determination
typescript
function determineStockStatus(spare: Spare): StockStatus {
  if (spare.rob === 0) {
    return 'ZERO';          // Out of stock - CRITICAL
  }
  
  if (spare.min === null || spare.min === 0) {
    return 'NOT_SET';       // Minimum not configured
  }
  
  if (spare.rob < spare.min) {
    return 'LOW';           // Below minimum - needs ordering
  }
  
  return 'OK';              // Adequate stock
}
Status Values: "ZERO" | "LOW" | "OK" | "NOT_SET"

2. Criticality Level Determination
typescript
function determineCriticality(
  spare: Spare,
  jobs: Job[],
  jobComponentLinks: JobComponentLink[],
  components: Component[]
): CriticalityLevel {
  
  // Find all jobs that require this spare part
  const jobsRequiringSpare = jobs.filter(job => {
    if (!job.requiredSpareParts) return false;
    
    // requiredSpareParts is JSONB array: [{ partCode, partName, quantity }]
    const spareParts = Array.isArray(job.requiredSpareParts) 
      ? job.requiredSpareParts 
      : [];
    
    return spareParts.some(sp => 
      sp.partCode === spare.partCode || 
      sp.partName === spare.partName  // Fallback match
    );
  });
  
  // For each job, find linked components
  let linkedToCriticalEquipment = false;
  const criticalComponentsList: string[] = [];
  
  for (const job of jobsRequiringSpare) {
    // Find component links for this job
    const links = jobComponentLinks.filter(link => link.jobId === job.id);
    
    for (const link of links) {
      const component = components.find(c => c.id === link.componentId);
      
      if (component && component.critical === true) {
        linkedToCriticalEquipment = true;
        criticalComponentsList.push(component.name);
      }
    }
  }
  
  // Determine criticality level
  if (linkedToCriticalEquipment) {
    return 'CRITICAL';       // Spare is for critical equipment
  }
  
  if (spare.rob === 0 || (spare.rob < spare.min && spare.min > 0)) {
    return 'ESSENTIAL';      // Not critical equipment but stock is low/zero
  }
  
  return 'NORMAL';           // Regular spare with adequate stock
}
Criticality Values: "CRITICAL" | "ESSENTIAL" | "NORMAL"

3. Shortage Quantity Calculation
typescript
function calculateShortage(spare: Spare): number {
  if (spare.min === null || spare.min === 0) {
    return 0;  // Cannot calculate shortage if min not set
  }
  
  return Math.max(0, spare.min - spare.rob);
}
4. Related Jobs & Components Extraction
typescript
function getRelatedJobsAndComponents(
  spare: Spare,
  jobs: Job[],
  jobComponentLinks: JobComponentLink[],
  components: Component[]
): {
  relatedJobs: string[];           // Array of job titles
  criticalComponents: string[];    // Array of critical component names
  linkedToCriticalEquipment: boolean;
  department: string | null;       // Primary department
} {
  
  const jobsRequiringSpare = jobs.filter(job => {
    const spareParts = Array.isArray(job.requiredSpareParts) 
      ? job.requiredSpareParts 
      : [];
    return spareParts.some(sp => 
      sp.partCode === spare.partCode || 
      sp.partName === spare.partName
    );
  });
  
  const relatedJobs = jobsRequiringSpare.map(j => j.jobTitle);
  const criticalComponents: string[] = [];
  const departments = new Set<string>();
  
  for (const job of jobsRequiringSpare) {
    if (job.department) departments.add(job.department);
    
    const links = jobComponentLinks.filter(link => link.jobId === job.id);
    
    for (const link of links) {
      const component = components.find(c => c.id === link.componentId);
      
      if (component) {
        if (component.critical === true) {
          criticalComponents.push(component.name);
        }
        if (component.department) departments.add(component.department);
      }
    }
  }
  
  return {
    relatedJobs: [...new Set(relatedJobs)],  // Remove duplicates
    criticalComponents: [...new Set(criticalComponents)],
    linkedToCriticalEquipment: criticalComponents.length > 0,
    department: departments.size > 0 ? Array.from(departments)[0] : null
  };
}
5. Remarks Auto-Generation
typescript
function generateRemarks(
  spare: Spare,
  stockStatus: StockStatus,
  criticality: CriticalityLevel,
  shortage: number
): string {
  
  if (stockStatus === 'ZERO') {
    if (criticality === 'CRITICAL') {
      return '⚠️ CRITICAL - OUT OF STOCK - Immediate procurement required for critical equipment';
    }
    return '⚠️ OUT OF STOCK - Immediate procurement required';
  }
  
  if (stockStatus === 'LOW') {
    if (criticality === 'CRITICAL') {
      return `⚠️ CRITICAL LOW STOCK - Shortage: ${shortage} units - Urgent ordering for critical equipment`;
    }
    return `Low stock - Shortage: ${shortage} units - Order recommended`;
  }
  
  if (stockStatus === 'NOT_SET') {
    return 'ℹ️ Minimum stock level not configured - Please set minimum threshold';
  }
  
  if (criticality === 'CRITICAL') {
    return 'Stock adequate - Critical equipment spare';
  }
  
  return 'Stock adequate';
}
