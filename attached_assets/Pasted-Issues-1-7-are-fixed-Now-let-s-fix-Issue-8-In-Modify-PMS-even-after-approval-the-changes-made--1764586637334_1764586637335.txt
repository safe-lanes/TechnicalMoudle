   // Apply the approved changes to the target entity
   if (changeRequest.changeData && changeRequest.targetId) {
     try {
       const changes = typeof changeRequest.changeData === 'string' 
         ? JSON.parse(changeRequest.changeData) 
         : changeRequest.changeData;
       
       console.log('[CR_APPROVE] Applying changes to', changeRequest.targetType);
       
       switch(changeRequest.targetType) {
         case 'component':
           await storage.updateComponent(changeRequest.targetId, changes);
           break;
         case 'spare':
           await storage.updateSpare(parseInt(changeRequest.targetId), changes);
           break;
         case 'job':
           await storage.updateJob(changeRequest.targetId, changes);
           break;
       }
       console.log('[CR_APPROVE] Changes applied successfully');
     } catch (applyError) {
       console.error('[CR_APPROVE] Failed to apply changes:', applyError);
     }
   }
