### 1.2 Feature Flag / Dual-Write Mode

#### Environment Variables

```bash
# File storage only (default)
STORAGE_MODE=file

# PostgreSQL storage only
STORAGE_MODE=postgres

# Dual-write mode (writes to both, reads from postgres)
STORAGE_MODE=dual
```

#### Enable Dual-Write for Validation

```bash
# Step 1: Set environment variable
export STORAGE_MODE=dual

# Step 2: Restart application
npm run dev

# Step 3: Run validation in background
npx tsx scripts/dual-write-validator.ts --monitor --duration=24h

# Step 4: Check validation results
cat logs/dual-write-validation.json
```

#### Switch to Postgres-Only

```bash
# After validation passes
export STORAGE_MODE=postgres
npm run dev
```

### Migration Script

```typescript
// scripts/migrations/01-core-reference.ts
import { db } from "../../server/db";
import { users, fleets, vessels, pmsVesselSettings } from "@shared/schema";
import { PersistentFileStorage } from "../../server/persistentStorage";

export async function migrateCoreReference() {
  const fileStorage = new PersistentFileStorage();
  const data = fileStorage.getAllData();
  
  console.log("Migrating Core Reference Data...");
  
  await db.transaction(async (tx) => {
    // 1. Migrate fleets (no dependencies)
    const fleetRecords = Object.values(data.fleets);
    console.log(`  Migrating ${fleetRecords.length} fleets...`);
    for (const fleet of fleetRecords) {
      await tx.insert(fleets).values({
        id: fleet.id,
        code: fleet.code,
        name: fleet.name,
        description: fleet.description,
        isActive: fleet.isActive ?? true,
      }).onConflictDoNothing();
    }
    
    // 2. Migrate vessels (depends on fleets)
    const vesselRecords = Object.values(data.vessels);
    console.log(`  Migrating ${vesselRecords.length} vessels...`);
    for (const vessel of vesselRecords) {
      await tx.insert(vessels).values({
        id: vessel.id,
        name: vessel.name,
        code: vessel.code,
        fleetId: vessel.fleetId,
        imoNumber: vessel.imoNumber,
        vesselType: vessel.vesselType,
        flag: vessel.flag,
        isActive: vessel.isActive ?? true,
      }).onConflictDoNothing();
    }
    
    // 3. Migrate users (depends on vessels for Ship role)
    const userRecords = Object.values(data.users);
    console.log(`  Migrating ${userRecords.length} users...`);
    for (const user of userRecords) {
      await tx.insert(users).values({
        id: user.id,
        username: user.username,
        password: user.password,
        fullName: user.fullName,
        email: user.email,
        role: user.role,
        vesselId: user.vesselId,
        department: user.department,
        isActive: user.isActive ?? true,
      }).onConflictDoNothing();
    }
    
    // 4. Migrate PMS vessel settings
    const settingsRecords = Object.values(data.pmsVesselSettings || {});
    console.log(`  Migrating ${settingsRecords.length} PMS settings...`);
    for (const settings of settingsRecords) {
      await tx.insert(pmsVesselSettings).values({
        vesselId: settings.vesselId,
        calendarLeadDaysCritical: settings.calendarLeadDaysCritical ?? 7,
        calendarLeadDaysNonCritical: settings.calendarLeadDaysNonCritical ?? 14,
        calendarGraceMode: settings.calendarGraceMode ?? 'COMPANY_STANDARD',
        calendarGraceDays: settings.calendarGraceDays ?? 7,
        rhLeadHoursCritical: settings.rhLeadHoursCritical ?? 50,
        rhLeadHoursNonCritical: settings.rhLeadHoursNonCritical ?? 100,
        rhGraceHours: settings.rhGraceHours ?? 168,
        locationAName: settings.locationAName ?? 'Location A',
        locationBName: settings.locationBName ?? 'Location B',
        updatedBy: settings.updatedBy || 'migration',
      }).onConflictDoNothing();
    }
  });
  
  console.log("Core Reference Data migration complete.");
}
```

### Post-Migration Validations

```typescript
// scripts/post-checks/01-core-reference.ts
export async function postMigrationValidations() {
  const fileStorage = new PersistentFileStorage();
  const fileData = fileStorage.getAllData();
  
  // Count validation
  const fileCounts = {
    users: Object.keys(fileData.users).length,
    fleets: Object.keys(fileData.fleets).length,
    vessels: Object.keys(fileData.vessels).length,
    pmsVesselSettings: Object.keys(fileData.pmsVesselSettings || {}).length,
  };
  
  const pgCounts = {
    users: (await db.select({ count: sql`count(*)` }).from(users))[0].count,
    fleets: (await db.select({ count: sql`count(*)` }).from(fleets))[0].count,
    vessels: (await db.select({ count: sql`count(*)` }).from(vessels))[0].count,
    pmsVesselSettings: (await db.select({ count: sql`count(*)` }).from(pmsVesselSettings))[0].count,
  };
  
  const countMatch = 
    fileCounts.users === Number(pgCounts.users) &&
    fileCounts.fleets === Number(pgCounts.fleets) &&
    fileCounts.vessels === Number(pgCounts.vessels) &&
    fileCounts.pmsVesselSettings === Number(pgCounts.pmsVesselSettings);
  
  // Sample data spot check
  const sampleUserId = Object.keys(fileData.users)[0];
  const fileUser = fileData.users[sampleUserId];
  const [pgUser] = await db.select().from(users).where(eq(users.id, Number(sampleUserId)));
  const sampleMatch = fileUser.username === pgUser?.username;
  
  // Referential integrity
  const orphanedUsers = await db.execute(sql`
    SELECT u.id FROM users u 
    LEFT JOIN vessels v ON u.vessel_id = v.id 
    WHERE u.vessel_id IS NOT NULL AND v.id IS NULL
  `);
  
  return {
    countMatch,
    fileCounts,
    pgCounts,
    sampleMatch,
    orphanedRecords: orphanedUsers.rows.length,
    passed: countMatch && sampleMatch && orphanedUsers.rows.length === 0
  };
}
```

```typescript
function sortByHierarchyDepth(components: any[]): any[] {
  const idToComponent = new Map(components.map(c => [c.id, c]));
  const depths = new Map<string, number>();
  
  function getDepth(id: string): number {
    if (depths.has(id)) return depths.get(id)!;
    const comp = idToComponent.get(id);
    if (!comp || !comp.parentId) {
      depths.set(id, 0);
      return 0;
    }
    const depth = getDepth(comp.parentId) + 1;
    depths.set(id, depth);
    return depth;
  }
  
  components.forEach(c => getDepth(c.id));
  return components.sort((a, b) => (depths.get(a.id) || 0) - (depths.get(b.id) || 0));
}
```

```typescript
// scripts/migrations/17-audit-log.ts
export async function migrateAuditLog() {
  const fileStorage = new PersistentFileStorage();
  const data = fileStorage.getAllData();
  
  const auditRecords = data.auditLogs || [];
  console.log(`Migrating ${auditRecords.length} audit log entries...`);
  
  // Batch insert for performance (100 records per batch)
  const batchSize = 100;
  for (let i = 0; i < auditRecords.length; i += batchSize) {
    const batch = auditRecords.slice(i, i + batchSize);
    await db.insert(auditLog).values(batch.map(record => ({
      timestamp: new Date(record.timestamp),
      userId: record.userId,
      vesselCode: record.vesselCode,
      componentCode: record.componentCode,
      entityType: record.entityType,
      entityId: record.entityId,
      actionType: record.actionType,
      fieldName: record.fieldName,
      oldValue: record.oldValue,
      newValue: record.newValue,
      source: record.source || 'migration',
      payload: record.payload,
    }))).onConflictDoNothing();
  }
  
  console.log("Audit Log migration complete.");
}
```
