### SQL DDL (for reference)

```sql
-- User roles enum
CREATE TYPE user_role AS ENUM ('Ship', 'Office', 'PMS Admin');

-- Users table
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
  username TEXT NOT NULL UNIQUE,
  password TEXT NOT NULL,
  full_name TEXT NOT NULL,
  email TEXT,
  role user_role NOT NULL DEFAULT 'Ship',
  vessel_id TEXT,
  department TEXT,
  is_active BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_users_role ON users(role);
CREATE INDEX IF NOT EXISTS idx_users_vessel ON users(vessel_id);

-- Fleets table
CREATE TABLE IF NOT EXISTS fleets (
  id TEXT PRIMARY KEY,
  code TEXT NOT NULL UNIQUE,
  name TEXT NOT NULL,
  description TEXT,
  is_active BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Vessels table
CREATE TABLE IF NOT EXISTS vessels (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  code TEXT NOT NULL,
  fleet_id TEXT,
  imo_number TEXT,
  vessel_type TEXT,
  flag TEXT,
  is_active BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_vessels_fleet ON vessels(fleet_id);
CREATE INDEX IF NOT EXISTS idx_vessels_imo ON vessels(imo_number);

-- PMS Vessel Settings table
CREATE TABLE IF NOT EXISTS pms_vessel_settings (
  id INTEGER PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
  vessel_id TEXT NOT NULL UNIQUE,
  calendar_lead_days_critical INTEGER NOT NULL DEFAULT 7,
  calendar_lead_days_non_critical INTEGER NOT NULL DEFAULT 14,
  calendar_grace_mode TEXT NOT NULL DEFAULT 'COMPANY_STANDARD',
  calendar_grace_days INTEGER NOT NULL DEFAULT 7,
  rh_lead_hours_critical INTEGER NOT NULL DEFAULT 50,
  rh_lead_hours_non_critical INTEGER NOT NULL DEFAULT 100,
  rh_grace_hours INTEGER NOT NULL DEFAULT 168,
  location_a_name TEXT NOT NULL DEFAULT 'Location A',
  location_b_name TEXT NOT NULL DEFAULT 'Location B',
  updated_by TEXT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_pms_settings_vessel_id ON pms_vessel_settings(vessel_id);
```

## Rollback Procedure

If migration needs to be undone:

```bash
# Revert to file storage
export STORAGE_MODE=file
npm run dev

# Optional: Clear PostgreSQL tables (CAUTION: Data loss)
npx tsx -e "
const { getDb } = require('./server/db');
const { users, fleets, vessels, pmsVesselSettings } = require('./shared/schema');

(async () => {
  const db = await getDb();
  
  console.log('Clearing Module 1 tables...');
  await db.delete(pmsVesselSettings);
  await db.delete(users);
  await db.delete(vessels);
  await db.delete(fleets);
  console.log('Done.');
})();
"
```

---

**Important Notes:**
- The migration is idempotent - rerunning will not overwrite existing rows
- Records are only inserted if they don't already exist (checked by primary key)
- To update existing records, delete them first or implement optional update support
